\section{Strong convexity and gradient monotonicity}
\begin{lemma}\label{lem:monotone}
Let $U:H\to\mathbb R$ be differentiable on a real Hilbert space and
$m$-strongly convex on $H$. Then, for all $x,y\in H$,
\[
  m\|x-y\|^2\le
  \langle\nabla U(x)-\nabla U(y),x-y\rangle.
\]
\end{lemma}
\begin{proof}
Strong convexity is equivalent to convexity of
$V(x)=U(x)-\tfrac m2\|x\|^2$. The supporting-plane inequality for $V$,
using $\nabla V(x)=\nabla U(x)-mx$, gives
\[
 U(y)\ge U(x)+\langle\nabla U(x),y-x\rangle
                 +\frac m2\|y-x\|^2.
\]
Apply this inequality also with $x$ and $y$ interchanged and add the
two inequalities. The function values cancel, leaving the claim.
\end{proof}
