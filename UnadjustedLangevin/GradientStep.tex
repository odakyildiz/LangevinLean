\section{The deterministic gradient step}
\begin{lemma}\label{lem:step}
Let $x,y,g_x,g_y\in H$, $h\ge0$, $L\ge0$, and $m\in\mathbb R$.
Suppose
\[
 m\|x-y\|^2\le\langle g_x-g_y,x-y\rangle,
 \qquad \|g_x-g_y\|\le L\|x-y\|.
\]
Writing $q=1-2hm+h^2L^2$, we have
\[
 \|(x-hg_x)-(y-hg_y)\|\le\sqrt q\,\|x-y\|.
\]
Here and below the square root is the nonnegative square root for
nonnegative arguments, extended by $\sqrt q=0$ for $q<0$, as in Lean.
\end{lemma}
\begin{proof}
Set $\delta=x-y$ and $g=g_x-g_y$. Expand the squared norm and use
the two hypotheses:
\begin{align*}
 \|\delta-hg\|^2
 &=\|\delta\|^2-2h\langle g,\delta\rangle+h^2\|g\|^2\\
 &\le\|\delta\|^2-2hm\|\delta\|^2+h^2L^2\|\delta\|^2
 =q\|\delta\|^2.
\end{align*}
Taking square roots gives the result. This is valid even without an
explicit assumption $q\ge0$: if $\delta\ne0$, the squared-norm
inequality forces $q\ge0$; if $\delta=0$, the Lipschitz bound forces
$g=0$ and both sides of the claim vanish.
\end{proof}
