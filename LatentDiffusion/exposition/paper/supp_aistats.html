   5 \subsection{Derivation of the Training Objective}\label{app:reparam_diffusion}
   6 
   7 We derive the training objective, the tilted free energy $\tilde{F}(\theta, \phi, q^{1:M})$ in \eqref{eq:free-energy}  and include a derivation of the standard reparametrized diffusion objective $\mathcal{L}_D(\theta, z_0)$ analogous to \citet{hodenoisingdiffusionprobabilistic2020, pmlr-v37-sohl-dickstein15}. For convenience, we re-state the objective for a single data point here and omit the superscript $m$:
   8 \begin{equation}\label{eq:app_free_energy}
   9     \tilde{F}(\theta, \phi, q) := \mathbb{E}_{q(z_{0:K})}\left[ \log \frac{q(z_{0:K})}{p_{\theta, \phi}( x, z_{0:K})} \right].
  10 \end{equation}
  11 
  12 \paragraph{Negative Lower Bound.} To see that $-\tilde{F}(\theta, \phi, q)$ is a lower bound of the log-likelihood $\log p_{\theta, \phi}(x)$, we first note that the tilted free energy in \eqref{eq:app_free_energy} can be re-written as:
  13 \begin{align*}
  14     \tilde{F}(\theta, \phi, q) &= \mathbb{E}_{q(z_0) q(z_{1:K}|z_0)}\left[ \log \frac{q(z_0) q(z_{1:K}|z_0)}{p_{\theta, \phi}(x|z_{0})p_\theta(z_{1:K}|z_0)p_\theta(z_0)} \right] \\
  15     &= \mathbb{E}_{q(z_0)}\left[ \mathbb{E}_{q(z_{1:K}|z_0)}\left[ \log \frac{q(z_0)}{p_{\theta, \phi}(x, z_0)} + \log \frac{q(z_{1:K}|z_0)}{p_\theta(z_{1:K}|z_0)} \right] \right] \\
  16     &= \mathbb{E}_{q(z_0)}\left[ \log \frac{q(z_0)}{p_{\theta, \phi}(x, z_0)} \right] + \mathbb{E}_{q(z_0)}[\KL(q(z_{1:K}|z_0) \| p_\theta(z_{1:K}|z_0))],
  17 \end{align*}
  18 where we have assumed the independence structure of the decoder $p_{\phi}( x|z_{0:K})=p_{\phi}( x|z_0)$; the first term is the usual free energy $F(\theta, \phi, q)=\mathbb{E}_{q(z_0)}\left[ \log {q(z_0)}-\log{p_{\theta, \phi}(x, z_0)} \right]$ in \eqref{eq:fen78awfnye8awnfeua} and the second term is non-negative. 
  19 It is thus easy to see that the tilted free energy $\tilde{F}(\theta, \phi, q)$ is obtained by replacing $p_{\theta,\phi}(x^m,z_0)$ in~\eqref{eq:fen78awfnye8awnfeua} with the tilted $\tilde{p}_{\theta,\phi}(x^m,z_0):=p_{\theta,\phi}(x^m,z_0)\exp(-\mathcal{R}(\theta,z_0))$, where $\mathcal{R}(\theta, z_0)= \KL\big(q(z_{1:K}|z_{0}) \|p_{\theta}(z_{1:K}|z_{0})\big)$. 
  20 
  21 Using Jensen's inequality, we see the negative of the first term is an upper bound of $-\log p_{\theta, \phi}(x)$:
  22 \begin{align*}
  23     \mathbb{E}_{q(z_0)}\left[ -\log \frac{p_{\theta, \phi}(x, z_0)}{q(z_0)} \right] \geq -\log \mathbb{E}_{q(z_0)}\left[ \frac{p_{\theta, \phi}(x, z_0)}{q(z_0)} \right] = -\log p_{\theta, \phi}(x),
  24 \end{align*}
  25 which leads to the sequence of inequalities:
  26 \begin{equation}
  27     \tilde{F}(\theta, \phi, q) \geq F(\theta, \phi, q) \geq -\log p_{\theta, \phi}(x).
  28 \end{equation}
  29 
  30 
  31 An alternative form of the objective $\tilde{F}(\theta, \phi, q)$ amenable to computation can be obtained by decomposing it as follows:
  32 \begin{align}
  33     \tilde{F}(\theta, \phi, q) &= \int \underbrace{\int \log \frac{q(z_{1:K}|z_{0})}{{p_{\theta}(z_{0:K})}} q(z_{1:K}|z_{0}) \dd z_{1:K} \ }_{\text{Diffusion Objective}  \ \mathcal{L}_D(\theta, z_0)}q(z_{0})\dd z_{0} - {\int \log \frac{{p_{\phi}( x|z_{0})}}{q(z_{0})} q(z_{0}) \ \dd z_{0}}.\label{eq_app:decomposed_obj}
  34 \end{align}

  37 \subsubsection{Derivation of the Diffusion Objective}
  38 The diffusion objective $\mathcal{L}_D(\theta, z_0)$ can be re-written as:
  39 \begin{align}
  40     \mathcal{L}_D(\theta, z_0) 
  41     &=\int\left(-\log p_K(z_K)+\log \frac{q_1(z_1|z_0)}{p_{\theta, 1} (z_0|z_1)}+\sum_{k=2}^K \log \frac{q_k(z_k| z_{k-1})}{p_{\theta, k}(z_{k-1}|z_k)}\right)q(z_{1:K}|z_{0})\dd z_{1:K}.\label{eq:app_R}
  42 \end{align}
  43 We can re-write $q_k(z_k|z_{k-1})$ using Bayes' rule as:
  44 \begin{align*}
  45 q_k(z_{k}|z_{k-1})
  46 &=q_k(z_{k-1}|z_{k},z_0)\frac{q(z_{k}|z_0)}{q(z_{k-1}|z_0)}.
  47 \end{align*}
  48 
  49 Combined with the Markov assumption $q(z_{1:K}|z_{0})=q(z_K|z_0)\prod_{k=2}^K q_k(z_{k-1}| z_{k}, z_{0})$,  we can rewrite \eqref{eq:app_R} above as:
  50 \begin{align}
  51     \mathcal{L}_D(\theta, z_0) 
  52     &= \mathbb{E}_{q(z_{1:K}|z_0)}\left[ \KL(q(z_K|z_0) \| p(z_K)) - \log p_{\theta, 1} (z_0|z_1) \right]  \\
  53     &+ \mathbb{E}_{q(z_{1:K}|z_0)}\left[\sum_{k=2}^K \KL ({q_k(z_{k-1}| z_{k}, z_0)}\|{p_{\theta, k}(z_{k-1}|z_k)}) \right]
  54 \end{align}
  55 \paragraph{Gaussian Transition Kernel.}
  56 For the forward process, we take
  57 \begin{equation}
  58     q_k(z_k| z_{k-1})=\mathcal{N}(z_k; \sqrt{1-\beta_k}z_{k-1}, \beta_k I),
  59 \end{equation}
  60 where $\{\beta_k\}_{k=1}^K$ is a linear noise schedule with $\beta_k = (1-k/K)\beta_0 + (k/K)\beta_K$. Using the notation $\alpha_{k}:=1-\beta_{k}$ and $\bar{\alpha}_{k}:=\prod_{j=1}^{k}\alpha_{j}$, we can derive the $k$-step transition kernel:
  61 \begin{equation}
  62     q(z_k|z_0) = \mathcal{N}(z_k; \sqrt{\bar{\alpha}_k}z_0, (1-\bar{\alpha}_k)I), \quad \forall k=1,\ldots, K.
  63 \end{equation}
  64 By Bayes' rule, we can compute:
  65 \begin{align}
  66     q_k(z_{k-1} | z_{k}, z_0) &= \mathcal{N}(z_{k-1}; \tilde{\mu}_k(z_k, z_0), \beta_k I), \\
  67     \tilde{\mu}_{k}(z_k,z_0) &= \frac{\sqrt{\bar{\alpha}_{k-1}}\beta_{k}}{1-\bar{\alpha}_{k}}{z}_{0}+\frac{\sqrt{\alpha_{k}}(1-\bar{\alpha}_{k-1})}{1-\bar{\alpha}_{k}}{z}_{k}.
  68 \end{align}
  69 For the backward process, we set:
  70 \begin{align}
  71     p_K(z_K) &= \mathcal{N}(0, I) \\
  72     p_{\theta, k}(z_{k-1}|z_{k})&=\mathcal{N}(z_{k-1}; \mu_{\theta, k}(z_k), \beta_k I) \quad \forall k=2, \ldots, K \\
  73     p_{\theta, 1}(z_{0}|z_1) &= \mathcal{N}(z_{k-1}; \mu_{\theta, k}(z_k),I)
  74 \end{align}
  75 Using the formula for the Kullback-Leibler divergence between isotropic Gaussian distributions, we obtain:
  76 \begin{align}
  77         \KL ({q_k(z_{k-1}| z_{k}, z_0)}\|{p_{\theta, k}(z_{k-1}|z_k)}) &=\frac{1}{2}\log\left(\frac{1-\bar{\alpha}_k}{{\beta}_k}\right)-\frac{d_z}{2}+\frac{d_z{\beta}_k+\|\tilde{\mu}_{k}(z_k,z_0)-\mu_{\theta, k}(z_k)\|^2}{2 (1-\bar{\alpha}_k)} \\
  78         \KL({q(z_K|z_0)}\|{p_\theta(z_K)})&=-\log(1-\bar{\alpha}_K)+\frac{d_z(1-\bar{\alpha}_K)^2-d_z+\bar{\alpha}_K\|{z_0}\|^2}{2} \label{eq_app:extra_term} \\
  79          -\log(p_\theta(z_0|z_1))&=\frac{d_z}{2}\log(2\pi (1-\bar{\alpha}_1))+\frac{\|z_0-\mu_{\theta, 1}(z_1)\|^2}{2(1-\bar{\alpha}_1)}\label{eq_app:discrete_gaussian}.
  80 \end{align}
  81 Note that in contrast to the standard pixel-space \gls*{DDPM}, we have the extra term in \eqref{eq_app:extra_term} depending on the latent $z_0$; we also use a Gaussian distribution with identity variance for $p_\theta (z_0|z_1)$ in \eqref{eq_app:discrete_gaussian} instead of the independent discrete decoder from \citet{hodenoisingdiffusionprobabilistic2020}. 
  82 
  83 \paragraph{Reparametrization.} We adopt the same $\epsilon$-prediction reparameterization as in \citet{hodenoisingdiffusionprobabilistic2020} to write:
  84 \[
  85 \mu_{\theta, k}({z}_k) = \tilde{\mu}_k \left( {z}_k, \frac{1}{\sqrt{\alpha_k}} \left( {z}_k - \sqrt{1 - \bar{\alpha}_k} \, \epsilon_{\theta, k}({z}_k) \right) \right) 
  86 = \frac{1}{\sqrt{\alpha_k}} \left( {z}_k - \frac{\beta_k}{\sqrt{1 - \bar{\alpha}_k}} \epsilon_{\theta, k}({z}_k) \right),
  87 \]
  88 which gives:
  89 \begin{align}
  90         \KL ({q_k(z_{k-1}| z_{k}, z_0)}\|{p_{\theta, k}(z_{k-1}|z_k)}) &=\frac{1}{2}\log\left(\frac{1-\bar{\alpha}_k}{{\beta}_k}\right)-\frac{d_z}{2}+\frac{d_z{\beta}_k+\beta_k\|\epsilon-\epsilon_{\theta, k}(z_k)\|^2}{2 \alpha_k (1-\bar{\alpha}_k)},
  91 \end{align}
  92 where $\epsilon \sim \mathcal{N}(0, I)$ is a random vector in $\R^{d_z}$ sampled independently from a standard Gaussian. Additionally, using the forward $k$-step transition kernel, we can write $z_k = z_k(z_0, \epsilon)=\sqrt{\bar{\alpha}_k}z_0+\sqrt{1-\bar{\alpha}_k}\epsilon$. Up to a constant term independent of $\theta$, we have the loss function $\mathcal{L}_D(\theta, z_0; \epsilon)\approx \hat{\mathcal{L}}(\theta,z_0;\epsilon, k)$, where we sample $k \sim \text{Unif}(1, \ldots, K)$ and $\epsilon \sim \mathcal{N}(0, I)$. We thus have the following loss:
  93 \begin{align}\label{app_eq:diffusion_likelihood}
  94     \hat{\mathcal{L}}_D(\theta,z_0;\epsilon, k):=\left\{
  95     \begin{array}{ll}\frac{1}{2}\|{z_0-\mu_{\theta,1}(z_1(z_0,\epsilon))\|}^2 + \frac{\bar{\alpha}_K\|{z_0\|}^2}{2}&\text{if }k=1\\
  96     \frac{\beta_k}{2\alpha_k(1-\bar{\alpha}_k)}\|{\epsilon-\epsilon_{\theta,k}(z_k(z_0,\epsilon))\|}^2 + \frac{\bar{\alpha}_K\|{z_0\|}^2}{2}&\text{if }2\leq k\leq K\\
  97     \end{array}
  98     \right.
  99 \end{align}
 100 In practice, we follow the approach in \citet{hodenoisingdiffusionprobabilistic2020} to use a simplified version of the diffusion objective:
 101 \begin{align}\label{app:one_sample_simplified_diffusion}
 102     \hat{\mathcal{L}}_{\text{simple}}(\theta, z_0; \epsilon)\approx \hat{\mathcal{L}}_{\text{simple}}(\theta,z_0;\epsilon, k):=\left\{
 103     \begin{array}{ll}\frac{1}{2}\|{z_0-\mu_{\theta,1}(z_1(z_0,\epsilon))\|}^2 + \frac{\bar{\alpha}_K\|{z_0\|}^2}{2}&\text{if }k=1\\
 104     \|{\epsilon-\epsilon_{\theta,k}(z_k(z_0,\epsilon))\|}^2 + \frac{\bar{\alpha}_K\|{z_0\|}^2}{2}&\text{if }2\leq k\leq K\\
 105     \end{array}
 106     \right.
 107 \end{align}
 108 More specifically, we draw a minibatch of indices $\mathcal{B} \subseteq [M]$, standard Gaussian random variables $\epsilon^{m,n} \sim \mathcal{N}(0,I)$, and $t^{m,n} \sim \text{Unif}(1, \ldots, K)$ for $m\in \mathcal{B}, n\in [N]$ to approximate the loss by:
 109 \begin{equation}\label{eq_app:simplified_diffusion}
 110     \hat{\mathcal{L}}_{\text{simple}}(\theta, z_0^{\mathcal{B},1:N}) = \frac{1}{|\mathcal{B}|N} \sum_{(m,n)\in \mathcal{B}\times [N]} \hat{\mathcal{L}}_{\text{simple}}(\theta, z_0^{m,n}; \epsilon^{m,n}, t^{m,n}).
 111 \end{equation}
 112 To simplify the notation, we will denote the one-sample approximation  $\hat{\mathcal{L}}_{\text{simple}}(\theta, z_0^{m,n}; \epsilon^{m,n}, t^{m,n})$ with $\hat{\mathcal{L}}_{\text{simple}}(\theta, z_0^{m,n})$ as in \Cref{sec:pratical-alg} and we use the same simplified notation in the rest of the appendix.

 115 \subsection{The Euclidean-Wasserstein Geometry on \texorpdfstring{$\R^{d_{\theta}} \times \R^{d_\phi}\times \mathcal{P}_2(\R^{d_z})^{M}$}{Theta x Phi x P2(Rdz)^M}}
 116 \label{app:otto_calc}
 117 To obtain the gradient flow in \Cref{sec:gradient_flow}, we view the product space of parameters and probability distributions $\R^{d_{\theta}} \times \R^{d_\phi}\times \mathcal{P}_2(\R^{d_z})^M$ as a Riemannian manifold. We will equip this product space with suitable tangent spaces and Riemannian metrics, which enable us to define the gradients and perform optimization. We omit the subscript on $z_0$ and denote it by $z$ for simplicity throughout this section.
 118 
 119 \paragraph{Tangent spaces.} Using the same set-up as in \citet{kuntz23a}, we concatenate the parameters as $\vartheta:=(\theta,\phi)\in \R^D$, where $D=d_\theta+d_\phi$, and assume the approximate posterior is a distribution with strictly positive density w.r.t. the Lebesgue measure and support $\sZ=\R^{d_z}$. We let the product manifold be $\mathcal{M}^{1:M}:=\R^D \times \mathcal{P}_2(\sZ)^M$. For each point $(\vartheta, q^{1:M}) \in \mathcal{M}^{1:M}$, we can define the tangent space $T\mathcal{M}^{1:M}$ and its dual $T^*\mathcal{M}^{1:M}$ as:
 120 \begin{align*}
 121     T_{(\vartheta, q^{1:M})}\mathcal{M}^{1:M} &= T_\vartheta\R^D \times \prod_{m\in[M]} T_{q^{m}}\mathcal{P}_2(\mathsf{Z})\\
 122     T_{(\vartheta, q^{1:M})}^*\mathcal{M}^{1:M} &= T_\vartheta^*\R^D \times \prod_{m\in[M]} T_{q^{m}}^*\mathcal{P}_2(\mathsf{Z})
 123 \end{align*}
 124 where we note that $T_\vartheta \R^D \cong T_\vartheta^* \R^D \cong \R^{D}$. For each $q^m \in \mathcal{P}_2(\sZ)$, we also define the tangent and cotangent space of $\mathcal{P}_2(\sZ)$ at $q^m$ as in \citet{otto31012001}:
 125 \begin{align*}
 126     T_{q^m}\mathcal{P}_2(\mathsf{Z}) &:= \left\{
 127     r:\mathsf{Z} \to \R: \int r(z) \dd z=0
 128     \right\} \\
 129     T_{q^m}^* \mathcal{P}_2(\mathsf{Z}) &:= \left\{
 130     f: \sZ \to \R
 131     \right\} / \R
 132 \end{align*}
 133 where the cotangent space $T_{q^m}^*\mathcal{P}_2(\mathsf{Z})$ is identified with the space of equivalence classes of functions that differ by an additive constant. 
 134 
 135 
 136 Furthermore, we define the \textit{duality pairing}, which is a triplet $(T_{(\vartheta, q^{1:M})}, T_{(\vartheta, q^{1:M})}^*, \langle\cdot,\cdot\rangle)$ with $$\langle\cdot,\cdot\rangle: T_{(\vartheta, q^{1:M})}\mathcal{M}^{1:M} \times T_{(\vartheta, q^{1:M})}^*\mathcal{M}^{1:M} \to \R$$ being a bilinear map that is the sum of the Euclidean inner product on the parameter space and the duality pairing on the Wasserstein-2 space:
 137 \begin{align*}
 138     \langle (\tau, r^{1:M}), (v, f^{1:M}) \rangle &:= \langle \tau, v \rangle + \sum_{m\in[M]} \langle r^m, f^m \rangle,
 139 \end{align*}
 140 where the Wasserstein-2 duality pairing is $(T_{q^m}\mathcal{P}_2(\sZ),T_{q^m}^*\mathcal{P}_2(\sZ), \langle\cdot,\cdot\rangle)$ with $\langle\cdot,\cdot\rangle: T_{q^m}\mathcal{P}_2(\sZ) \times T_{q^m}^*\mathcal{P}_2(\sZ) \to \R$ given by:
 141 \begin{equation}
 142     \langle r^m,f^m\rangle:=\int f^m(z)r^m(z) \dd z, \quad \forall r^m \in T_{q^m}\mathcal{P}_2(\sZ), f^m\in T_{q^m}^*\mathcal{P}_2(\sZ).
 143 \end{equation}
 144 
 145 \paragraph{The metric.}\label{app:metric} We can thus equip the manifold $\mathcal{M}^{1:M}$ with Riemannian metric $g=(g_{(\vartheta, q^{1:M})})_{(\vartheta,q^{1:M})\in \mathcal{M}^{1:M}}$
 146 defined as:
 147 $$
 148 g_{(\vartheta, q^{1:M})}((\tau, r^{1:M}), (\tau', r'^{1:M})) := \langle (\tau, r^{1:M}), G_{(\vartheta, q^{1:M})}(\tau', r'^{1:M}) \rangle, \quad \forall (\vartheta, q^{1:M}) \in \mathcal{M}^{1:M},
 149 $$
 150 where $G_{(\vartheta, q^{1:M})}: T\mathcal{M}^{1:M} \to T^*\mathcal{M}^{1:M}$ is an invertible, self-adjoint, and positive-definite linear map. We will only consider tensors in a block-diagonal form (cf. \citet[Chapter~2]{lee2018riemannian}), which defines the metric via:
 151 \begin{equation}
 152     \langle (\tau, r^{1:M}), G_{(\vartheta, q^{1:M})}(\tau', r'^{1:M}) \rangle = \langle \tau, \mathrm{G}_{\vartheta} \tau' \rangle + \frac{1}{M}\sum_{m\in[M]} \langle r^m, \mathbf{G}_{q^m}^W r'^m \rangle  \quad \forall (\tau, r^{1:M}) \in T\mathcal{M}^{1:M}, \ (\vartheta, q^{1:M}) \in \mathcal{M}^{1:M}.
 153 \end{equation}
 154 Here we take $\mathrm{G}_{\vartheta}: T\R^D \to T^* \R^D$ as the identity map (which corresponds to the usual Euclidean metric) and $\mathbf{G}_{q}^W: T\mathcal{P}_2(\mathsf{Z}) \to T^*\mathcal{P}_2(\mathsf{Z})$ to be the tensor for Wasserstein-2 distance on $\mathcal{P}_2(\mathsf{Z})$ defined through its inverse:
 155 \begin{equation}
 156     (\mathbf{G}_q^W)^{-1}f:=-\nabla_{z} \cdot (q \nabla_{z}f), \quad \forall f\in C^\infty(\mathcal{P}_2(\mathsf{Z})).
 157 \end{equation}
 158 When the context is clear, we will also write $\langle\tau, \tau'\rangle_{\vartheta} =\langle \tau, \mathrm{G}_{\vartheta} \tau' \rangle$ and $\langle r^m, r'^m \rangle_{q^m} = \langle r^m, \mathbf{G}_{q^m}^W r'^m \rangle$.
 159 
 160 
 161 \paragraph{The gradient.} To perform gradient descent on manifolds, we further need an analogue of the gradient for a smooth function $F: \mathcal{M}^{1:M}\to \R$ as in the Euclidean space. This is a vector field $\nabla F: \mathcal{M}^{1:M}\to T\mathcal{M}^{1:M}$ satisfying:
 162 \begin{equation}
 163     \begin{aligned}
 164     g_{(\vartheta, q^{1:M})} (\nabla F(\vartheta, q^{1:M}), (\tau, r^{1:M}))
 165     &= \lim_{t \to 0} \frac{F(\vartheta + t\tau, q^{1:M} + tr^{1:M}) - F(\vartheta, q^{1:M})}{t} \\
 166     &\quad \forall (\tau, r^{1:M}) \in T\mathcal{M}^{1:M},\; (\vartheta, q^{1:M}) \in \mathcal{M}^{1:M}.
 167     \end{aligned}
 168 \end{equation}
 169 By expressing in local coordinates, we can compute the gradient as:
 170 \begin{equation}\label{eq:app_inv_tensor_grad}
 171     \nabla F(\vartheta, q^{1:M}) = G_{(\vartheta, q^{1:M})}^{-1} \delta F(\vartheta, q^{1:M}), \quad \forall (\vartheta, q^{1:M})\in \mathcal{M}^{1:M},
 172 \end{equation}
 173 where $\delta F: \mathcal{M}^{1:M}\to T^*\mathcal{M}^{1:M}$ is the first variation of $F$ defined as the unique cotangent vector field satisfying:
 174 \begin{equation}\label{eq:app_first_variation}
 175     \langle (\tau, r^{1:M}), \delta F(\vartheta, q^{1:M}) \rangle = \lim_{t \to 0} \frac{F(\vartheta + t\tau, q^{1:M} + tr^{1:M}) - F(\vartheta, q^{1:M})}{t} 
 176 \quad \forall (\tau, r^{1:M}) \in T\mathcal{M}^{1:M},\; (\vartheta, q^{1:M}) \in \mathcal{M}^{1:M}.
 177 \end{equation}
 178 \paragraph{The distance function.} To define the distance on the manifold $\mathcal{M}^{1:M}$, we recall that the Riemannian distance function is defined as the length of the minimizing geodesics $\gamma: [0,1]\to \mathcal{M}^{1:M}$ between $(\vartheta, q^{1:M})$ and $(\vartheta', q'^{1:M})$:
 179 \begin{align}
 180     \mathsf{d}_{\mathcal{M}^{1:M}}((\vartheta, q^{1:M}), (\vartheta', q'^{1:M}))^2 := \inf_{\gamma} \int_0^1 g_{(\vartheta_t,q_t^{1:M})}(\dot{\gamma}(t), \dot{\gamma}(t)) \dd t
 181     &= \|\vartheta-\vartheta'\|_2^2 + \frac{1}{M}\sum_{m=1}^M \mathsf{d}_{W_2}(q^m, (q')^m)^2,
 182 \end{align}
 183 where the second equality follows from~\citet[Section 4.3]{otto31012001} and $W_2$ is the Wasserstein-2 distance on $\mathcal{P}_2(\sZ)$.

 185 \subsection{Derivation of the Gradient Flow}
 186 \label{app:grad_deriv}
 187 Equipped with the tools above, we now derive the gradients of the averaged free energy $\tilde{F}(\theta, \phi, q^{1:M})=M^{-1}\sum_{m=1}^M \tilde{F}^m(\theta, \phi, q^m)$, where $\tilde{F}^m$ is the single-datapoint free energy for $x^m$. We first compute the single-datapoint first variation \eqref{eq:app_first_variation}, and then lift it componentwise to the averaged objective:
 188 \begin{lemma}\label{app:lemma_1st_var}
 189      Given the free energy of the form in \eqref{eq_app:decomposed_obj}:
 190 \begin{equation}
 191     \tilde{F}(\theta, \phi, q)  =  \int \mathcal{L}_D(\theta, z) q(z)\dd z - \int \log \left(\frac{p_\phi( x|z)}{q(z)}\right) q(z)\dd z.
 192 \end{equation}
 193 The first variation $\delta \tilde{F}(\theta, \phi, q)=(\delta_q \tilde{F}(\theta, \phi, q), \delta_\theta \tilde{F}(\theta, \phi, q), \delta_\phi \tilde{F}(\theta, \phi, q))$ is given by:
 194 \begin{align}
 195     \delta_q \tilde{F}(\theta, \phi, q) &= \log \left(\frac{q(z)}{p_\phi( x|z)}\right) + \mathcal{L}_D(\theta, z)\label{eq_app:1st_var_q}   \\
 196     \delta_\theta \tilde{F}(\theta, \phi, q) &= \int \nabla_\theta \mathcal{L}_D(\theta, z) q(z) \dd z\label{eq_app:1st_var_theta}  \\
 197     \delta_\phi \tilde{F}(\theta, \phi, q) &= - \int \nabla_\phi \log p_\phi( x|z) q(z) \dd z\label{eq_app:1st_var_phi} 
 198 \end{align}
 199 \end{lemma}
 200 \begin{proof}
 201     For \cref{eq_app:1st_var_theta,eq_app:1st_var_phi}, it suffices to note that by grouping the parameters into $\tilde\theta := (\theta, \phi) \in \Theta=\R^{D}$, we can compute by Taylor expansion:
 202 \begin{align*}
 203         \tilde{F}(\tilde{\theta} + t\tau, q) &= \tilde{F}(\tilde{\theta},q) + \int f(\tilde{\theta} + t\tau, z) q(z) \dd z -\int f(\tilde{\theta}, z) q(z) \dd z \\
 204 &= \tilde{F}(\tilde{\theta},q) + \int [t \langle \tau, \nabla_{\tilde{\theta}} f(\tilde{\theta}, z) \rangle + o(t)] q(z) \dd z \\
 205 &= \tilde{F}(\tilde\theta, q) + t \left\langle \tau, \int \nabla_{\tilde{\theta}} f(\tilde{\theta}, z) q(z) \dd z \right\rangle + o(t)
 206 \end{align*}
 207 where we have set $f(\tilde\theta, z):= \mathcal{L}_D(\theta, z) - \log p_\phi( x|z)$ and used $f(\tilde\theta + t\tau, z) = f(\tilde\theta, z) + t \langle \tau, \nabla_{\tilde\theta} f(\tilde\theta, z) \rangle + o(t)$. 
 208 
 209 For \eqref{eq_app:1st_var_q}, we recall the first variation is linear, thus it suffices to compute $\delta_q \int \mathcal{L}_D(\theta, z) q(z)\dd z$ and $\delta_q\int (\log {q(z)} - \log {p_\phi( x|z)}) q(z)\dd z$ separately. Now note that:
 210 \begin{align*}
 211     \int \mathcal{L}_D(\theta, z) [q(z)+tr(z)]\dd z &= \int \mathcal{L}_D(\theta, z) q(z)\dd z + t\int \mathcal{L}_D(\theta, z) r(z) \dd z
 212 \end{align*}
 213 and for the latter we note that $\log(z + t)(z + t) = \log(z)z + [\log(z) + 1]t + o(t)$, from which it follows:
 214 \begin{align*}
 215     \int \log \left(\frac{q(z)+tr(z)}{p_\phi( x|z)}\right) [q(z)+tr(z)]\dd z 
 216     &= \int (\log q(z)) q(z)+ [\log q(z) + 1] tr(z) +o(t) \dd z \\
 217     &- \int \log p_\phi( x|z) q(z) \dd z - t\int \log p_\phi( x|z) r(z) \dd z \\
 218     &= \int \log \left(\frac{q(z)}{p_\phi( x|z)}\right) q(z)\dd z \\&+ t \int \log \left(\frac{q(z)}{p_\phi( x|z)}\right) r(z) \dd z + o(t)
 219 \end{align*}
 220 where we have used that $\int r(z)\dd z=0$ for all $r\in T\mathcal{P}_2(\mathsf{Z})$.
 221 \end{proof}

 223 \begin{proposition}
 224     Under the normalized product geometry of \Cref{app:metric}, the gradients of the averaged free energy $\tilde{F}(\theta, \phi, q^{1:M})$ with respect to $(\theta, \phi, q^{1:M})$ are given by:
 225     \begin{align}
 226     \nabla_\theta \tilde{F}(\theta, \phi, q^{1:M}) &= \frac{1}{M}\sum_{m=1}^M \int \left[\nabla_{\theta} \mathcal{L}_D(\theta, z)\right] q^m(z) \dd z \\
 227     \nabla_\phi \tilde{F}(\theta, \phi, q^{1:M}) &= -\frac{1}{M}\sum_{m=1}^M \int \left[\nabla_{\phi} \log p_{\phi}( x^m | z)\right] q^m(z) \dd z \\
 228     \nabla_{q^m} \tilde{F}(\theta, \phi, q^{1:M}) &=  \nabla_{z} \cdot \left[
 229         q^m(z) \nabla_{z} \left[
 230             \log \left(
 231                 \frac{p_{\phi}( x^m|z)}{q^m(z)}
 232             \right) - \mathcal{L}_D(\theta, z)
 233         \right]
 234     \right], \quad \forall m\in[M].
 235 \end{align}
 236 \end{proposition}
 237 \begin{proof}
 238     Recall that on the manifold $\mathcal{M}^{1:M}$, the gradient can be computed from the first variation via the metric tensor \eqref{eq:app_inv_tensor_grad}:
 239     \begin{equation*}
 240         \nabla_{(\tilde{\theta},q^{1:M})} \tilde{F}(\tilde{\theta},q^{1:M}) = G_{(\tilde{\theta}, q^{1:M})}^{-1} \delta \tilde{F}(\tilde{\theta}, q^{1:M}),
 241     \end{equation*}
 242     where $\tilde{\theta}=(\theta, \phi)$. For the parameter block, the metric is Euclidean, so the $\theta$- and $\phi$-gradients are the corresponding averaged first variations. For each distribution $q^m$, we have
 243     \[
 244         \delta_{q^m}\tilde{F}(\theta,\phi,q^{1:M}) = \frac{1}{M}\delta_q \tilde{F}^m(\theta,\phi,q^m),
 245     \]
 246     where $\delta_q \tilde{F}^m$ is given by \Cref{app:lemma_1st_var} with $x$ replaced by $x^m$ and $q$ by $q^m$. Since the inverse metric on the $m$th Wasserstein block is $M(\mathbf{G}_{q^m}^W)^{-1}$, the factor $M^{-1}$ in $\delta_{q^m}\tilde{F}$ is cancelled, yielding
 247     \[
 248         \nabla_{q^m}\tilde{F}(\theta,\phi,q^{1:M}) = (\mathbf{G}_{q^m}^W)^{-1}\delta_q \tilde{F}^m(\theta,\phi,q^m).
 249     \]
 250     Recalling that $(\mathbf{G}_q^W)^{-1}f=-\nabla_{z} \cdot (q \nabla_{z}f)$ for all $f\in C^\infty(\mathcal{P}_2(\mathsf{Z}))$ gives the desired result.
 251 \end{proof}

 258 \subsubsection{KL Divergence Annealing} 
 259 \label{app:kl_anneal_deriv}
 260 We note that annealing the KL divergence with $\gamma_t$ is equivalent to applying the same weighting to the gradient of the entropy and the prior terms. In our case, we modify the gradient flow in (\ref{eq:theta_ode}--\ref{eq:flow}) as follows:
 261 \begin{align}
 262     \nabla_\theta \tilde{F}(\theta_t, \phi_t, q_t^{1:M}) &= \frac{1}{M}\sum_{m=1}^M\mathbb{E}_{q_t^m(z_{0,t})}\left[\nabla_{\theta} \gamma_t\mathcal{L}_D(\theta_t, z_{0})\right], \\
 263     \nabla_\phi \tilde{F}(\theta_t, \phi_t, q_t^{1:M}) 
 264     &=-\frac{1}{M}\sum_{m=1}^M\mathbb{E}_{q_t^m(z_{0,t})}\left[\nabla_{\phi} \log p_{\phi_t}( x^m | z_{0,t})\right], \\
 265     \nabla_{q^m} \tilde{F}(\theta_t, \phi_t, q_t^{1:M}) &= \nabla_{z_{0}} \cdot [
 266         q^m(z_{0,t}) \nabla_{z_{0}}
 267             \log 
 268                 p_{\phi_t}( x^m|z_{0,t})] \\
 269     &- \gamma_t \nabla_{z_{0}}\cdot[q^m(z_{0,t}) \nabla_{z_0} [\log {q^m(z_{0,t})}
 270              - \mathcal{L}_D(\theta_t, z_{0,t})
 271         ]
 272     ],  \quad\forall m\in [M]
 273 \end{align}
 274 which correspond to the Fokker-Planck equation of the following system of SDEs:
 275 \begin{align}
 276 \dd {\theta}_t &= -\frac{1}{M}\sum_{m=1}^M\mathbb{E}_{q^m_t(z_0)}\left[\nabla_{\theta} \gamma_t\mathcal{L}_D(\theta_t, Z_{0,t}^m)\right] \dd t, \\
 277     \dd {\phi}_t &= \frac{1}{M}\sum_{m=1}^M\mathbb{E}_{q^m_t(z_0)}\left[\nabla_{\phi} \log p_{\phi_t}( x^m | Z_{0,t}^m)\right] \dd t, \\
 278     \dd Z_{0,t}^m &= \nabla_{z_{0}} [\log(p_{\phi_t}( x^m|Z_{0,t}^m))-\gamma_t\mathcal{L}_D(\theta_t,Z_{0,t}^m)]dt+\sqrt{2 \gamma_t} \dd W_{t}^m, \quad\forall m\in [M]
 279 \end{align}
 280 Thus, we only need to adjust the amount of noise injected to the in \eqref{eq:disc_par} by setting $\sqrt{2h_z}$ to $\sqrt{2h_z \gamma_t}$ and weigh the diffusion loss $\mathcal{L}_D(\theta, z_0)$ by $\gamma_t$. Using the KL annealing scheme and re-weighted diffusion loss \eqref{app:one_sample_simplified_diffusion} discussed in \Cref{sec:pratical-alg}, we thus modify the loss in \eqref{eq:batch_loss} with:
 281 \begin{equation}
 282         \hat{\mathcal{L}}(\theta_t,\phi_t,z^{1:M,1:N}_{0,t}) := \frac{1}{N |\mathcal{B}|} \sum_{(m,n)\in  \mathcal{B} \times [N]} \left[
 283     \gamma_t\hat{\mathcal{L}}_{\text{simple}}(\theta_t, z_{0,t}^{m,n}) - \log p_\phi ( x^m | z_{0,t}^{m,n})
 284     \right],\label{eq:loss_kl_anneal}
 285 \end{equation}
 286 where $\hat{\mathcal{L}}_{\text{simple}}$ is the simplified diffusion loss in Appendix~\ref{app:reparam_diffusion} and we take $\gamma_t = c_{KL}\min (1, t/T)$ for $T$ the number of KL warm-up steps and $c_{KL}$ a constant coefficient.

 289 \subsubsection{Preconditioning and Momentum}
 290 To avoid flat minima and ill-conditioning, we use an adaptive version of Langevin dynamics similar to \citet{Li2016,kim2020stochasticgradientlangevindynamics} based on Adam \citep{kingma2017adammethodstochasticoptimization}. In particular, we update each particle by running:
 291 \begin{align}\label{eq:adam_vz_update}
 292 z_{0,t+1}^{m,n} &\gets z_{0,t}^{m,n} + h_z G_{t}^{m,n} M_{t}^{m,n} +\sqrt{\frac{2h_z}{M}} (G_{t}^{m,n})^{1/2}W_{t}^{m,n}, \quad \forall (m,n) \in [M] \times [N]
 293 \end{align}
 294 where $h_z$ is the stepsize of particle updates, and we compute the moment $M_{t}^{m,n}$  and the preconditioner $G_{t}^{m,n}$ by:
 295 \begin{subequations}\label{eq:vz_momentum}
 296 \begin{align}
 297 	M_{t}^{m,n} &\gets a M_{t-1}^{m,n} + {MN}(1-a) \nabla_{z_{0}^{m,n}}
 298 \hat{\mathcal{L}}(\theta_t,\phi_t,z^{1:M,1:N}_{0,t}) \\
 299 	V_{t}^{m,n} &\gets b V_{t-1}^{m,n} + (1-b) \text{diag}\left( \left[{MN}\nabla_{z_{0}^{m,n}  }
 300 \hat{\mathcal{L}}(\theta_t,\phi_t,z^{1:M,1:N}_{0,t})\right]^{\otimes 2}\right)  \\
 301 			G_{t}^{m,n} &\gets (V_{t}^{m,n} + \epsilon I)^{-1/2}
 302 \end{align}
 303 \end{subequations}
 304 Here $\hat{\mathcal{L}}$ is defined as in \eqref{eq:loss_kl_anneal}, $\epsilon >0$ is a positive constant to avoid numerical instabilities, $0<a,b<1$ are hyperparameters like in Adam \citep{kingma2017adammethodstochasticoptimization} (we take $a=0.9, b=0.999$ following the default), and we used the notation $\text{diag}(v^{\otimes 2})$ to denote the diagonal matrix with the $(i,j)^{th}$ entry being $v_i^2\delta_{ij}$. In practice, we set the preconditioner on the noise in \eqref{eq:adam_vz_update} to $G_{t}^{m,n}=I$ during the first epoch for numerical stability. We also scale the noise by $|\mathcal{B}|^{-1}$ in implementation to prevent the noise from dominating the gradient. We now present the full algorithm for training.
 305 \begin{algorithm}[htbp]
 306 \caption{IPLD with adaptive Langevin dynamics}\label{algo:train_1}
 307 \begin{algorithmic}[1]
 308 \State \textbf{Inputs:} Training data points $\{ x^m\}_{m\in [M]}$,   optimizers $\texttt{Opt}_{\theta}, \texttt{Opt}_{\phi}$, Step sizes $h_\theta, h_\phi, h_z$, KL weights $\gamma_t$, Initial particles $\{z_{0,0}^{m,n}\}_{(m,n) \in [M] \times [N]}$ sampled from $\mathcal{N}(0,I)$, Initial parameters $\phi,\theta$.
 309 \While{not converged}
 310     \State Sample a mini-batch of indices $\mathcal{B}\subset [M]$
 311     \State Compute the loss $\hat{\mathcal{L}}(\theta_t,\phi_t,z^{1:M,1:N}_{0,t})$ as in \eqref{eq:loss_kl_anneal}
 312     \For{$(m,n)\in \mathcal{B}\times[M]$}
 313         \Comment{Update the particle cloud}
 314 	\State Compute momentum $M_{t}^{m,n}$ and preconditioner $G_{t}^{m,n}$ as in \eqref{eq:vz_momentum}
 315         \State Sample independent $W_{t,m}^i$ for $(i,m)\in [N]\times[M]$
 316         \If{$m\in \mathcal{B}$}
 317 	\State $z_{0,t+1}^{m,n} \gets z_{0,t}^{m,n} + h_z G_{t}^{m,n} M_{t}^{m,n}$ 
 318     \EndIf
 319     \State $z_{0,t+1}^{m,n} \gets z_{0,t}^{m,n} +\sqrt{\frac{2h_z \gamma_t}{M}} (G_{t}^{m,n})^{1/2}W_{t}^{m,n}$ 
 320     \EndFor
 321     \Comment{Update model parameters}
 322     \State $\theta_{t+1} \gets \texttt{Opt}_\theta(h_\theta, \theta_t, \hat{\mathcal{L}})$
 323     \State $\phi_{t+1} \gets \texttt{Opt}_\phi(h_\phi, \phi_t, \hat{\mathcal{L}})$
 324     \State $t \gets t+1$
 325 \EndWhile
 326 \State \textbf{return} $\theta_{t}, \phi_t, z_{0,t}^{1:M,1:N}$
 327 \end{algorithmic}
 328 \end{algorithm}

 344 \paragraph{Evaluation.} We report the \gls*{MMD} \citep{jmlr:v13:gretton12a} between the generated samples and the ground truth on the \gls*{GMM} dataset. For samples $\{x_i\}_{i=1}^m \sim P$ and $\{y_i\}_{j=1}^m \sim Q$, the unbiased Monte-Carlo estimate of \gls*{MMD} is defined as:
 345 \begin{equation*}
 346     \text{MMD}(P,Q) \approx \frac{1}{m(m-1)} \sum_{i=1}^{m} \sum_{\substack{j=1 \\ j \ne i}}^{m} k( x_i,  x_j)
 347 + \frac{1}{n(n-1)} \sum_{i=1}^{n} \sum_{\substack{j=1 \\ j \ne i}}^{n} k(y_i, y_j)
 348 - \frac{2}{mn} \sum_{i=1}^{m} \sum_{j=1}^{n} k( x_i, y_j),
 349 \end{equation*}

A total of $m=n=10,000$ samples were generated via the reverse process to compute the approximation of the \gls*{MMD}. We repeat the training runs with 50 different random seeds (thus different initializations) and report the mean and standard error. The error bars shown in \Cref{fig:ldm_gmm_comparison} are the 1-standard error $\hat{\sigma}/\sqrt{50}$, where the standard deviation $\hat{\sigma}$ is estimated using \texttt{numpy}'s default implementation.

 397 \paragraph{Notation and assumptions.} We denote ${\rho}_{\theta,\phi}(\cdot)$ as the unnormalized density for:
 398 \begin{equation}
 399     {p}_{\theta,\phi}(x,\cdot)=p_{\phi}(x|\cdot)p_\theta(\cdot).
 400 \end{equation}
 401 Similarly, we define $\tilde{\rho}_{\theta,\phi}(\cdot)$ as the unnormalized density for:
 402 \begin{equation}
 403     \tilde{p}_{\theta,\phi}(x,\cdot):=p_{\theta,\phi}(x,\cdot)\exp(-\mathcal{R}(\theta,\cdot)),
 404 \end{equation}
 405 where $p_{\theta,\phi}(x,z_0)=p_{\theta}(z_0) p_\phi(x|z_0)$ and $\mathcal{R}(\theta, z_0) := \KL(q(z_{1:K}|z_{0})||p_{\theta}(z_{1:K}|z_{0}))$. In multi-datapoint settings, we add a superscript to denote the dependence on $x^m$ i.e. 
 406 \begin{equation}
 407     \tilde\rho^m_{\theta,\phi}(\cdot):=\tilde p_{\theta,\phi}(x^m,\cdot).
 408 \end{equation}
 409 The $M$-datapoint version is defined as:
 410 \begin{equation}
 411     \tilde{\rho}_{\theta,\phi}(z^{1:M}):=\prod_{m=1}^M\rho^m_{\theta,\phi}(z^m)=\prod_{m=1}^M \tilde{p}_{\theta,\phi}(x^m,z^m).
 412 \end{equation}
 413 And the joint log density is:
 414 \begin{equation}
 415     \ell^m(\theta,\phi,z):=\log \rho^m_{\theta,\phi}(z)=\log p_{\phi}(x^m|z)+\log p_\theta(z).
 416 \end{equation}
 417 We also denote $\pi_{\theta,\phi}^m(\cdot):={p}_{\theta,\phi}(\cdot|x^m)$ as the normalized density and define similarly $\tilde \pi_{\theta,\phi}^m(\cdot):={\tilde p}_{\theta,\phi}(\cdot|x^m)$ for the tilted model.
 418 We further denote the normalizing constant of ${p}_{\theta,\phi}(x,\cdot)$ as $A_{\theta,\phi}$ and that of $\tilde{p}_{\theta,\phi}(x,\cdot)$ as $\tilde{A}_{\theta,\phi}$. 
 419 We recall the definition of the free energy $F(\theta, \phi,q)$ as:
 420 \begin{equation}
 421     {F}(\theta, \phi, q) := \mathbb{E}_{q(z_0)}\left[
 422    \log \frac{q(z_{0})}{p_{\theta, \phi}( x, z_{0})}
 423    \right].
 424 \end{equation}
 425 We set the modified free energy as:
 426 \begin{equation}
 427    \tilde{F}(\theta, \phi, q) := \mathbb{E}_{q(z_{0:K})}\left[ \log \frac{q(z_{0:K})}{p_{\theta, \phi}( x, z_{0:K})} \right]=F(\theta, \phi,q) + \mathbb{E}_{q(z_0)}\left[\mathcal{R}(\theta, z_0) 
 428    \right].
 429 \end{equation}
 430 
 431 The tilted free energy aggregated over multiple data points $\{x^m\}_{m=1}^M$ is defined as:
 432 \begin{equation}
 433     \tilde{F}(\theta, \phi, q^{1:M}) := \frac{1}{M}\sum_{m=1}^M \tilde{F}(\theta, \phi, q^m).
 434 \end{equation}
 435 
 436 For clarity, we drop the subscript on $z_0$ and instead write $z$ for the latent variable in subsequent sections and we use the notations $\Theta:=\R^{d_\theta}$, $\Phi:=\R^{d_\phi}$ for the parameter spaces and $\mathcal{P}_2(\sZ)$ for the space of probability measures over the latent space $\sZ=\R^{d_z}$, which we assume to have densities with respect to the Lebesgue measure. We define the product manifold and the distance $\mathsf{d}_{\mathcal{M}}$ as in \Cref{app:metric}:
 437 \begin{align}
 438     \mathcal{M} := \ \Theta \times \Phi \times \mathcal{P}_2(\sZ), \quad 
 439     \mathsf{d}_{\mathcal{M}}((\theta, \phi, q), (\theta', \phi', q')) := \sqrt{
 440     \|(\theta,\phi)-(\theta',\phi')\|^2 + \mathsf{d}_{W_2}(q, q')^2.
 441     }\label{eq:single_datapoint_prodmetric}
 442 \end{align}
 443 When $(\theta, \phi, q)$ are random variables, we overload the notation $\mathsf{d}$ to denote:
 444 \begin{align}
 445 \mathsf{d}((\theta, \phi, q), (\theta', \phi', q')) := \sqrt{
 446     \mathbb{E}[\|(\theta,\phi)-(\theta',\phi')\|^2] + \mathbb{E}[\mathsf{d}_{W_2}(q, q')^2].
 447     }\label{eq:random_variable_metric}
 448 \end{align}
 449 To extend the above to multiple datapoints, we set:
 450 \begin{align}
 451     \mathcal{M}^{1:M}&:= \ \Theta \times \Phi \times \mathcal{P}_2(\sZ)^M, \\
 452     \mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta', \phi', q'^{1:M})) &:= \sqrt{
 453     \|(\theta,\phi)-(\theta',\phi')\|^2 + \frac{1}{M}\sum_{m=1}^M \mathsf{d}_{W_2}(q^m, q'^m)^2.\label{eq:multi_datapoint_prodmetric}
 454     }
 455 \end{align}

We use a superscript $1$ for related product spaces (\emph{e.g.}, $\mathcal{M}^1$ and $\mathcal{M}^{1:M,1}$) to indicate the restriction to $\mathcal{P}_2^1(\sZ)$.

 467 \subsubsection{Extended Log-Sobolev Inequality}
 468 In this section, we show that strong log-concavity implies extended log-Sobolev inequality. We first define the $M$-datapoint version of the extended log-Sobolev inequality in \citet{caprioerrorboundsparticle2024}:
 469 
 470 \begin{definition}[\gls*{xlsi}]\label{def:xLSI}
 471     Denote $\tilde{F}_\star:=  \inf_{(\theta, \phi,q^{1:M})\in \mathcal{M}^{1:M,1}} \tilde{F}(\theta, \phi, q^{1:M})$ as the optimum of $\tilde{F}(\theta, \phi, q^{1:M})$. We say the measure $(\tilde{\rho}_{\theta,\phi}(\dd z^{1:M}))_{(\theta,\phi)\in\Theta\times\Phi}$ satisfies the extended log-Sobolev inequality with constant $\lambda>0$ if for all $(\theta, \phi,q^{1:M}) \in \ \mathcal{M}^{1:M,1}$ we have:
 472     \begin{equation}
 473         2\lambda[\tilde{F}(\theta, \phi, q^{1:M})-\tilde{F}_\star] \leq I(\theta, \phi, q^{1:M}),
 474     \end{equation}
 475     where we define $I(\theta, \phi, q^{1:M})$ as:
 476     \begin{equation}\label{eq:def_fisher_info}
 477         \begin{aligned}
 478             I(\theta, \phi, q^{1:M})
 479             &:= \|\nabla_{(\theta, \phi)} \tilde{F}(\theta, \phi, q^{1:M})\|^2 + \frac{1}{M}\sum_{m=1}^M \int \left\| \nabla_{z} \log \left( \frac{q^{m}(z)}{\tilde{p}_{\theta,\phi}(x^m,z)}\right)  \right\|^2 q^{m}(\dd z) \\
 480             &= \left\|
 481             \frac{1}{M}\sum_{m=1}^M \int \nabla_{(\theta, \phi)} \log \tilde{p}_{\theta, \phi}(x^m,z)\, q^{m}(\dd z)
 482             \right\|^2 + \frac{1}{M}\sum_{m=1}^M \int \left\| \nabla_{z} \log \left( \frac{q^{m}(z)}{\tilde{p}_{\theta,\phi}(x^m,z)}\right)  \right\|^2 q^{m}(\dd z).
 483         \end{aligned}
 484     \end{equation}
 485 \end{definition}
 486 
 487 Using the functional $I(\theta, \phi, q^{1:M})$ defined in~\eqref{eq:def_fisher_info}, we can state the following extension of de Bruijn's identity:
 488 \begin{proposition}[de Bruijn's Identity]\label{prop:de_bruijin}
 489     Under Assumption~\Cref{assump:solution_reg}, we have:
 490     \begin{equation}\label{eq:de_bruijin}
 491         \frac{\dd}{\dd t} \tilde{F}(\theta_t, \phi_t, q_t^{1:M}) = -I(\theta_t, \phi_t, q_t^{1:M}), \qquad \forall t>0.
 492     \end{equation}
 493 \end{proposition}
 494 \begin{proof}
 495     By definition,
 496     \begin{align*}
 497         \tilde{F}(\theta_t,\phi_t,q_t^{1:M})
 498         = \frac{1}{M}\sum_{m=1}^M \left[
 499             \int \log\left(\frac{q_t^m(z)}{\tilde{p}_{\theta_t,\phi_t}(x^m,z)}\right) q_t^m(\dd z)
 500         \right].
 501     \end{align*}
 502     Under Assumption~\Cref{assump:solution_reg}, we can differentiate under the integral sign and the flow satisfies
 503     \begin{align*}
 504         \frac{\partial_t q_t^m(z)}{\partial t} &= \nabla_z \cdot \left[
 505             q_t^m(z)\nabla_z \log \left(\frac{q_t^m(z)}{\tilde{p}_{\theta_t,\phi_t}(x^m,z)}\right)
 506         \right], \\
 507         (\dot{\theta}_t,\dot{\phi}_t) &= \frac{1}{M}\sum_{m=1}^M \int \nabla_{(\theta,\phi)} \log \tilde{p}_{\theta_t,\phi_t}(x^m,z)\, q_t^m(\dd z).
 508     \end{align*}
 509     Using integration by parts, for each $m$ we obtain
 510     \begin{align*}
 511         \frac{\dd}{\dd t} \int \log(q_t^m(z)) q_t^m(\dd z)
 512         &= \int (\log(q_t^m(z))+1)\, \frac{\partial_t q_t^m(z)}{\partial t}\dd z \\
 513         &= - \int \left\langle \nabla_z \log(q_t^m(z)), \nabla_z \log \left(\frac{q_t^m(z)}{\tilde{p}_{\theta_t,\phi_t}(x^m,z)}\right) \right\rangle q_t^m(\dd z)
 514     \end{align*}
 515     and
 516     \begin{align*}
 517         \frac{\dd}{\dd t} \int \log(\tilde{p}_{\theta_t,\phi_t}(x^m,z)) q_t^m(\dd z)
 518         &= \int \left\langle \nabla_{(\theta,\phi)} \log \tilde{p}_{\theta_t,\phi_t}(x^m,z), (\dot{\theta}_t,\dot{\phi}_t) \right\rangle q_t^m(\dd z) \\
 519         &\quad - \int \left\langle \nabla_z \log(\tilde{p}_{\theta_t,\phi_t}(x^m,z)), \nabla_z \log \left(\frac{q_t^m(z)}{\tilde{p}_{\theta_t,\phi_t}(x^m,z)}\right) \right\rangle q_t^m(\dd z).
 520     \end{align*}
 521     Therefore,
 522     \begin{align*}
 523         \frac{\dd}{\dd t} \tilde{F}(\theta_t,\phi_t,q_t^{1:M})
 524         &= - \|(\dot{\theta}_t,\dot{\phi}_t)\|^2 - \frac{1}{M}\sum_{m=1}^M \int \left\| \nabla_z \log \left(\frac{q_t^m(z)}{\tilde{p}_{\theta_t,\phi_t}(x^m,z)}\right) \right\|^2 q_t^m(\dd z) \\
 525         &= - I(\theta_t,\phi_t,q_t^{1:M}),
 526     \end{align*}
 527     where the last equality follows from the definition of $I$ in~\eqref{eq:def_fisher_info}.
 528 \end{proof}

 531 \begin{lemma}[Geodesics on $\mathcal{M}^{1:M}$]\label{lem:geodesic}
 532     A curve $\gamma(t) : t \in [0, 1] \mapsto \mathcal{M}^{1:M}$ is a geodesic if and only if $\gamma(t) = (\gamma_{\theta}(t), \gamma_\phi (t), \gamma_{q^1}(t), \ldots, \gamma_{q^M}(t))$, where $\gamma_{\theta}$ and $\gamma_\phi$ are geodesics in the Euclidean parameter space and each $\gamma_{q^m}$ is a geodesic in $(\mathcal{P}_2(\sZ), \mathrm{d}_{W_2})$. In particular, if $\gamma(t)$ is a geodesic in $\mathcal{M}^{1:M}$ connecting $(\theta, \phi, q^{1:M})$ and $(\theta', \phi', (q')^{1:M})$ then 
 533     \begin{align*}
 534         \gamma_{\theta}(t) &= (1-t)\theta + t\theta', \\
 535         \gamma_{\phi}(t) &= (1-t)\phi + t\phi', \\
 536         \gamma_{q^m}(t) &= (h_t)_{\#}\varrho^m, \quad m=1,\ldots,M,
 537     \end{align*}
 538     where $\varrho^m$'s are Wasserstein-2 optimal transport plans for $(q^m, (q')^m)$ and $h_t(z,z') = (1-t)z + tz'$. Furthermore, if $q^m$'s have densities with respect to the Lebesgue measure, then we can also write:
 539     \begin{align*}
 540         \gamma_{q^m}(t) = ((1-t)\mathrm{id} + t\nabla_z f^m)_{\#} q^m, \quad \forall m \in [M],
 541     \end{align*}
 542     for some convex function $f^m$.
 543 \end{lemma}
 544 \begin{proof}
 545     The first claim follows from the definition of the product metric~\eqref{eq:multi_datapoint_prodmetric}. The second claim is a result of the characterization of geodesics (cf.~\citet[Theorem 5.27]{santambrogio2015optimal}). The last claim follows from Brenier's theorem (cf.~\citet[Theorem 1.17]{santambrogio2015optimal}).
 546 \end{proof}

 548 \begin{lemma}[Geodesic convexity of $\tilde{F}$]\label{lem:geodesic_convexity}
 549     Under Assumption~\Cref{assump:strong_log_concave}, the tilted free energy $\tilde{F}(\theta, \phi, q^{1:M})$ is $\lambda$-geodesically convex on $\mathcal{M}^{1:M}$, that is, for any pair $(\theta, \phi, q^{1:M})$ and $(\theta', \phi', (q')^{1:M})$ in $\mathcal{M}^{1:M}$ and any geodesic $\gamma(t)$ connecting them, we have:
 550     \begin{align*}
 551         \tilde{F}(\gamma(t)) &\leq (1-t)\tilde{F}(\theta, \phi, q^{1:M}) + t\tilde{F}(\theta', \phi', (q')^{1:M}) 
 552         - \frac{\lambda t(1-t)}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta', \phi', (q')^{1:M}))^2.
 553     \end{align*}
 554 \end{lemma}
 555 \begin{proof}
 556     First recall that $\tilde{F}(\theta, \phi, q^{1:M})= M^{-1} \sum_{m=1}^M \mathbb{E}_{q^m(z)}[\log(q^m(z))-\tilde{\ell}^m(\theta,\phi,z)]$.
 557     We note that the negative entropy $q^m \mapsto \int \log(q^m(z)) q^m(\dd z)$ is geodesically convex on $\mathcal{P}_2(\sZ)$ (cf.~\citet[Theorem 7.28]{santambrogio2015optimal}) and thus the average $q^{1:M} \mapsto M^{-1}\sum_{m=1}^M \int \log(q^m(z)) q^m(\dd z)$ is geodesically convex on $\mathcal{P}_2(\sZ)^M$. By an argument similar to~\citet[Lemma 21]{caprioerrorboundsparticle2024}, we can show the map $V: (\theta, \phi, z) \mapsto -M^{-1}\sum_{m=1}^M \int \tilde{\ell}^m(\theta,\phi, z) q^m(\dd z)$ is $\lambda$-strongly convex along the geodesic $\gamma(t)$:
 558     \begin{align*}
 559         V(\gamma(t)) &= - \int \frac{1}{M}\sum_{m=1}^{M} \tilde{\ell}^m(\gamma_\theta(t), \gamma_\phi(t), z) \gamma_{q^m}(t)(\dd z) \\
 560         &=-\int \frac{1}{M}\sum_{m=1}^{M} \tilde{\ell}^m((1-t)\theta + t\theta', (1-t)\phi + t\phi', (1-t)z + tz') \varrho^m(\dd z, \dd z') \\
 561         &\leq -\int \frac{1}{M}\sum_{m=1}^{M} (1-t) \tilde{\ell}^m(\theta, \phi, z) + t\tilde{\ell}^m(\theta', \phi', z') \varrho^m(\dd z, \dd z') \\
 562         &\quad - \frac{\lambda t(1-t)}{2} \int \frac{1}{M}\sum_{m=1}^M \|(\theta, \phi, z) - (\theta', \phi', z')\|^2 \varrho^m(\dd z, \dd z') \\
 563         &=(1-t)V(\theta, \phi, q^{1:M}) + tV(\theta', \phi', (q')^{1:M}) - \frac{\lambda t(1-t)}{2} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta', \phi', (q')^{1:M}))^2,
 564     \end{align*}
 565     where we have used Lemma~\ref{lem:geodesic} in the second equality and the strong log-concavity in the inequality. The conclusion follows from the definition of the product metric~\eqref{eq:multi_datapoint_prodmetric}.
 566 \end{proof}
 567 We similarly provide an extension to~\citet[Lemma~22]{caprioerrorboundsparticle2024}:
 568 \begin{lemma}\label{lem:lower_deriv}
 569     Let $\gamma(t)$ be a geodesic in $\mathcal{M}^{1:M,1}$ connecting $(\theta, \phi, q^{1:M})$ and $(\theta', \phi', (q')^{1:M})$. Then we have:
 570     \begin{align*}
 571         \liminf_{t \to 0^+} \frac{\tilde{F}(\gamma(t)) - \tilde{F}(\gamma(0))}{t} &\geq \frac{1}{M} \sum_{m=1}^M \left\langle \dot{\gamma}_{q^m}(0), \nabla_{q^m} \tilde{F}(\theta, \phi, q^{m}) \right\rangle_{q^m} + \langle 
 572         (\theta', \phi') - (\theta, \phi), \nabla_{(\theta, \phi)} \tilde{F}(\theta, \phi, q^{1:M})
 573         \rangle
 574     \end{align*}
 575 \end{lemma}
 576 \begin{proof}
 577     The proof is an adaptation of~\citet[Lemma 22]{caprioerrorboundsparticle2024} by noting that the computation can be done separately for each $m$.
 578 \end{proof}
 579 \begin{theorem}[Strong log-concavity $\implies$ xLSI]\label{thm:sLC_implies_xLSI}
 580     Suppose Assumptions~\Cref{assump:model_reg} and~\Cref{assump:strong_log_concave} hold, then the family of measures $(\tilde{\rho}_{\theta,\phi}(\dd z^{1:M}))_{(\theta,\phi)\in\Theta\times\Phi}$ satisfies the extended log-Sobolev inequality with constant $\lambda>0$.
 581 \end{theorem}
 582 \begin{proof}
 583     The proof extends that of~\citet[Theorem 6]{caprioerrorboundsparticle2024} to the product manifold $\Theta \times \Phi \times \mathcal{P}_2(\sZ)^M$. Let $\gamma(t)$ be a geodesic in $\mathcal{M}^{1:M}$ connecting $(\theta, \phi, q^{1:M})$ and $(\theta', \phi', (q')^{1:M})$. By Lemma~\ref{lem:geodesic_convexity}, we have:
 584     \begin{align*}
 585         \liminf_{t \to 0^+} \frac{\tilde{F}(\gamma(t)) - \tilde{F}(\gamma(0))}{t} &\leq \tilde{F}(\theta', \phi', (q')^{1:M}) - \tilde{F}(\theta, \phi, q^{1:M}) - \frac{\lambda}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta', \phi', (q')^{1:M}))^2.
 586     \end{align*}
 587     Setting $(\theta', \phi', (q')^{1:M})=(\theta_\star, \phi_\star, (q_\star')^{1:M})$ to be a minimizer of $\tilde{F}$ and using the previous Lemma~\ref{lem:lower_deriv}, we obtain:
 588     \begin{align*}
 589         \tilde{F}(\theta, \phi, q^{1:M}) - \tilde{F}_\star &\leq -\frac{1}{M} \sum_{m=1}^M \left\langle \dot{\gamma}_{q^m}(0), \nabla_q { \tilde{F}(\theta, \phi, q^m)} \right\rangle_{q^m} - \langle 
 590         (\theta_\star, \phi_\star) - (\theta, \phi), \nabla_{(\theta, \phi)} \tilde{F}(\theta, \phi, q^{1:M})
 591         \rangle \\
 592         &\quad - \frac{\lambda}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta_\star, \phi_\star, q_\star^{1:M}))^2 \\
 593         &\leq \frac{1}{M} \sum_{m=1}^M \mathsf{d}_{W_2}(q^m, q_\star^m) \left\| \nabla_q {\tilde{F}(\theta, \phi, q^m)} \right\|_{q^m} + \|(\theta_\star, \phi_\star) - (\theta, \phi)\| \|\nabla_{(\theta, \phi)} \tilde{F}(\theta, \phi, q^{1:M})\| \\
 594         &\quad - \frac{\lambda}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta_\star, \phi_\star, q_\star^{1:M}))^2,
 595     \end{align*}
 596     where we have used the definition of the inner product on the tangent space $T_q\mathcal{P}_2^1(\sZ)$ and Lemma~\ref{lem:geodesic}:
 597     \begin{equation*}
 598         \|\dot{\gamma}_{q^m}(0)\|_{q^m}=\|\nabla_z f^m - \mathrm{id}\|_{L^2(q^m)}=\mathsf{d}_{W_2}(q^m, q_\star^m),
 599     \end{equation*}
 600     and the Cauchy-Schwarz inequality. One more application of the Cauchy-Schwarz inequality yields:
 601     \begin{align*}
 602         \tilde{F}(\theta, \phi, q^{1:M}) - \tilde{F}_\star &\leq \left[\frac{1}{M}\sum_{m=1}^M \mathsf{d}_{W_2}(q^m, q_\star^m)^2 + \|(\theta, \phi) - (\theta_\star, \phi_\star)\|^2\right]^{1/2} \\
 603         &\quad \times \left[\frac{1}{M}\sum_{m=1}^M \left\| \nabla_{q} { \tilde{F}(\theta, \phi, q^m)} \right\|_{q^m}^2 + \|\nabla_{(\theta, \phi)} \tilde{F}(\theta, \phi, q^{1:M})\|^2\right]^{1/2} \\
 604         &\quad - \frac{\lambda}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta_\star, \phi_\star, q_\star^{1:M}))^2 \\
 605         &= {\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta_\star, \phi_\star, q_\star^{1:M}))} \sqrt{I(\theta, \phi, q^{1:M})} - \frac{\lambda}{2}\mathsf{d}_{\mathcal{M}^{1:M}}((\theta, \phi, q^{1:M}), (\theta_\star, \phi_\star, q_\star^{1:M}))^2 \\
 606         &\leq \frac{1}{2\lambda} I(\theta, \phi, q^{1:M}),
 607     \end{align*}
 608     where we have upper bounded the first term using Young's inequality $ab \leq a^2/(2\lambda) + \lambda b^2/2$ in the last step.
 609 \end{proof}

 612 \subsubsection{Extended Talagrand-type Inequality}
 613 We now show that \gls*{xlsi} implies an extended Talagrand-type inequality. Throughout this section, we assume Assumptions~\Cref{assump:model_reg} and \Cref{assump:solution_reg} hold. We first provide the $M$-datapoint version of extended Talagrand in \citet{caprioerrorboundsparticle2024}.
 614 \begin{definition}[Extended Talagrand-type Inequality]\label{defn:xTalagrand}
 615     The family of measures $(\tilde{\rho}_{\theta,\phi}(\dd z^{1:M}))_{(\theta,\phi)\in\Theta\times\Phi}$ satisfy the extended Talagrand-type inequality with constant $\lambda>0$ if for all $(\theta, \phi,q^{1:M}) \in \ \mathcal{M}^{1:M}$:
 616     \begin{align}
 617         2[\tilde{F}(\theta, \phi, q^{1:M})-\tilde{F}_\star] &\geq \lambda\inf_{(\theta, \phi,q^{1:M}) \in \ \mathcal{M}^{1:M}} \mathsf{d}((\theta, \phi, q^{1:M}), \mathcal{M}^{1:M}_\star)^2
 618     \end{align}
 619     where $\mathcal{M}^{1:M}_\star:=\arg\min_{(\theta, \phi,q^{1:M}) \in \ \mathcal{M}^{1:M}} \tilde{F}(\theta, \phi, q^{1:M})$ is the optimal set of $\tilde{F}$.
 620 \end{definition}
 621 
 622  The proof hinges on a few auxiliary results adapted from \citet{caprioerrorboundsparticle2024}.
 623 \begin{lemma}\label{lem:distance_deriv}
 624     Under Assumption~\Cref{assump:solution_reg}, we have:
 625     \begin{equation*}
 626         \frac{\dd}{\dd t} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M})) \leq \sqrt{I(\theta_t, \phi_t, q_t^{1:M})}, \quad \forall t>0,
 627     \end{equation*}
 628     where $I(\theta, \phi, q^{1:M})$ is defined in~\eqref{eq:def_fisher_info}.
 629 \end{lemma}
 630 \begin{proof}
 631     For $(\theta, \phi, q^{1:M}) \in \mathcal{M}^{1:M}$, we define the velocity field $v_t^{1:M} = (v_t^1, \ldots, v_t^M)$ as:
 632     \begin{align*}
 633         v_t^m(z) = -\nabla_z \log \left( \frac{q_t^m(z)}{\tilde{\rho}_{\theta_t, \phi_t}^m(z)} \right), \quad \forall m \in [M].
 634     \end{align*}
 635     Using the proof to~\citet[Lemma 16]{caprioerrorboundsparticle2024}, for each $m$, we have:
 636     \begin{align*}
 637         \frac{\dd}{\dd t} \mathsf{d}_{W_2}(q_t^m, q^m)^2 \leq 2 \mathsf{d}_{W_2}(q_t^m, q^m)\sqrt{\int \|v_t^m(z)\|^2 q_t^m(\dd z)}.
 638     \end{align*}
 639     By the Cauchy-Schwarz inequality, we have:
 640     \begin{align*}
 641         \frac{\dd}{\dd t} \frac{1}{M} \sum_{m=1}^M \mathsf{d}_{W_2}(q_t^m, q^m)^2 &\leq \frac{2}{M} \sum_{m=1}^M \mathsf{d}_{W_2}(q_t^m, q^m)\sqrt{\int \|v_t^m(z)\|^2 q_t^m(\dd z)}.
 642     \end{align*}
 643     Combining this with the definition of the Euclidean counterpart:
 644     \begin{align*}
 645         \frac{\dd}{\dd t} \|(\theta_t, \phi_t) - (\theta, \phi)\|^2 = 2\left\langle \frac{\dd}{\dd t} (\theta_t, \phi_t), (\theta_t, \phi_t) - (\theta, \phi) \right\rangle \leq 2\left\|\nabla_{(\theta, \phi)} \tilde{F}(\theta_t, \phi_t, q_t^{1:M})\right\| \|(\theta_t, \phi_t) - (\theta, \phi)\|,
 646     \end{align*}
 647     we obtain with an application of the Cauchy-Schwarz inequality similar to~\Cref{thm:sLC_implies_xLSI}:
 648     \begin{align*}
 649         \frac{\dd}{\dd t}\frac{1}{2} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M}))^2 &\leq  \left(\frac{1}{M}\sum_{m=1}^M \mathsf{d}_{W_2}(q_t^m, q^m)^2\right)^{1/2}\left(\frac{1}{M}\sum_{m=1}^M \int \|v_t^m(z)\|^2 q_t^m(\dd z)\right)^{1/2} \\
 650         &\qquad + \|(\theta_t, \phi_t) - (\theta, \phi)\| \|\nabla_{(\theta, \phi)} \tilde{F}(\theta_t, \phi_t, q_t^{1:M})\| \\
 651         &\leq \sqrt{I(\theta_t, \phi_t, q_t^{1:M})} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M})),
 652     \end{align*}
 653     which implies the conclusion.
 654 \end{proof}

 656 \begin{lemma}\label{lem:lower_semi_continuous}
 657     The tilted free energy $\tilde{F}(\theta, \phi, q^{1:M})$ is lower semi-continuous on $\mathcal{M}^{1:M,1}$.
 658 \end{lemma}
 659 \begin{proof}
 660     We can re-write $\tilde{F}(\theta, \phi, q^{1:M}) = M^{-1}\sum_{m=1}^M {F}(\theta, \phi, q^m)+\mathbb{E}_{q^m}[\mathcal{R}(\theta, z)]$, where we recall $\mathcal{R}(\theta, z_0) = \KL\big(q(z_{1:K}|z_{0}) \|p_{\theta}(z_{1:K}|z_{0})\big) \geq 0$.
 661     We note that it suffices to show that $U: (\theta, q) \mapsto \mathbb{E}_{q}[\mathcal{R}(\theta, z)] = \int \mathcal{R}(\theta, z) q(\dd z)$ is lower semi-continuous on $\Theta \times \mathcal{P}_2(\sZ)$ (hence $\Theta \times \mathcal{P}_2^1(\sZ)$), since~\citet[Lemma 18]{caprioerrorboundsparticle2024} shows that the untilted free energy ${F}(\theta, \phi, q)$ is lower semi-continuous on $\mathcal{M}^1$. The conclusion thus follows from the definition of the space $(\mathcal{M}^{1:M,1}, \mathsf{d}_{\mathcal{M}^{1:M}})$.
 662 
 663     To show $U$ is lower semi-continuous, let $\{(\theta_n, q_n)\}_{n\in\mathbb{N}}$ be a sequence in $\Theta \times \mathcal{P}_2(\sZ)$ converging to some $(\theta_\infty, q_\infty) \in \Theta \times \mathcal{P}_2(\sZ)$ in the topology induced by $\mathsf{d}_{\mathcal{M}}$. The convergence in $\mathsf{d}_{W_2}$ implies weak convergence of $q_n \rightharpoonup q_\infty$ \citep{figalli2021invitation} and we have $\theta_n \to \theta_\infty$.
 664     Since $\mathcal{R}$ is non-negative and continuous in both arguments, by applying Portmanteau theorem on the product measures $\mu_n:=\delta_{\theta_n}(\dd \theta) \otimes q_n(\dd z)$,
 665     we have:
 666     \begin{align*}
 667         \liminf_{n\to\infty} U(\theta_n, q_n) &= \liminf_{n\to\infty} \int \mathcal{R}(\theta_n, z) \mu_n(\dd \theta, \dd z)\\
 668         &\geq\int \mathcal{R}(\theta, z) \mu_\infty(\dd \theta, \dd z) = U(\theta_\infty, q_\infty),
 669     \end{align*}
 670     which concludes the proof.
 671 \end{proof}

 673 \begin{lemma}\label{lem:cauchy}
 674     Denote $\mathcal{M}_\star := \arg\min_{(\theta, \phi, q^{1:M}) \in \mathcal{M}^{1:M}} \tilde{F}(\theta, \phi, q^{1:M})$ as the set of minimizers of $\tilde{F}$ and $\tilde{F}_\star := \inf_{(\theta, \phi, q^{1:M}) \in \mathcal{M}^{1:M}} \tilde{F}(\theta, \phi, q^{1:M})$ as the optimal value. Under the extended log-Sobolev inequality, for a Cauchy sequence $\{(\theta_n, \phi_n, q_n^{1:M})\}_{n\in \mathbb{N}}$ in $\mathcal{M}^{1:M,1}$, there exists an increasing sequence of $t_n \to +\infty$ such that:
 675     \begin{equation*}
 676         (\theta_{t_n}, \phi_{t_n}, q_{t_n}^{1:M}) \to (\theta_\star, \phi_\star, q_\star^{1:M}) \in \mathcal{M}_\star, \quad \text{as} \ n \to +\infty,
 677     \end{equation*}
 678     for some $(\theta_\star, \phi_\star, q_\star^{1:M}) \in \mathcal{M}_\star$ in the topology induced by $\mathsf{d}_{\mathcal{M}^{1:M}}$ on $\mathcal{M}^{1:M}$.
 679 \end{lemma}
 680 \begin{proof}
 681     Since both the Euclidean space and $\mathcal{P}_2(\sZ)$ are complete metric spaces \citep[Theorem 6.18]{villani2008optimal}, the product space $(\mathcal{M}^{1:M}, \mathsf{d}_{\mathcal{M}^{1:M}})$ is also complete. Therefore, the sequence $(\theta_\infty, \phi_\infty, q_\infty^{1:M})$ is also Cauchy in $\mathcal{M}^{1:M}$ and converges to some limit $(\theta_\infty, \phi_\infty, q_\infty^{1:M}) \in \mathcal{M}^{1:M}$. By the lower semi-continuity of $\tilde{F}$, we have:
 682     \begin{align*}
 683         \tilde{F}_\star \leq \tilde{F}(\theta_\infty, \phi_\infty, q_\infty^{1:M}) \leq \liminf_{n\to\infty} \tilde{F}(\theta_{t_n}, \phi_{t_n}, q_{t_n}^{1:M}) = \tilde{F}_\star,
 684     \end{align*}
 685     where the first inequality is by the optimality of $\tilde{F}_\star$ and the equality follows from the fact the exponential convergence induced by \gls*{xlsi} (cf. the rightmost inequality of~\Cref{thm:exp_convg_appendix} below).
 686 \end{proof}

 688 \begin{theorem}[\gls*{xlsi} $\implies$ \gls*{xt2i}]\label{thm:xLSI_implies_xTalagrand}
 689     Suppose the family of measures $(\tilde{\rho}_{\theta,\phi}(\dd z^{1:M}))_{(\theta,\phi)\in\Theta\times\Phi}$ satisfies the extended log-Sobolev inequality with constant $\lambda>0$. Then it also satisfies the extended Talagrand-type inequality with the same constant $\lambda>0$.
 690 \end{theorem}
 691 \begin{proof}
 692     Using Lemma~\ref{lem:distance_deriv} and the \gls*{xlsi}, an argument similar to that in~\citet[Theorem 4]{caprioerrorboundsparticle2024} yields the following sequence of inequalities for any $(\theta, \phi, q^{1:M}) \in \mathcal{M}^{1:M}$:
 693     \begin{align*}
 694         \frac{\dd}{\dd t} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M})) \leq \sqrt{I(\theta_t, \phi_t, q_t^{1:M})} &\leq \frac{I(\theta_t, \phi_t, q_t^{1:M})}{\sqrt{2\lambda[\tilde{F}(\theta_t, \phi_t, q_t^{1:M}) - \tilde{F}_\star]}}. 
 695     \end{align*}
 696     Using de Bruijn's identity~\eqref{eq:de_bruijin}, we have:
 697     \begin{align*}
 698         \frac{\dd}{\dd t} \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M})) &\leq -\frac{\dd}{\dd t} \sqrt{\frac{2[\tilde{F}(\theta_t, \phi_t, q_t^{1:M}) - \tilde{F}_\star]}{\lambda}}.
 699     \end{align*}
 700     For any interval $(t,t')$, integrating yields:
 701     \begin{align}
 702         &\mathsf{d}_{\mathcal{M}^{1:M}}((\theta_{t'}, \phi_{t'}, q_{t'}^{1:M}), (\theta, \phi, q^{1:M})) - \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t, \phi_t, q_t^{1:M}), (\theta, \phi, q^{1:M})) \nonumber \\
 703         &\leq \sqrt{\frac{2[\tilde{F}(\theta_t, \phi_t, q_t^{1:M}) - \tilde{F}_\star]}{\lambda}} - \sqrt{\frac{2[\tilde{F}(\theta_{t'}, \phi_{t'}, q_{t'}^{1:M}) - \tilde{F}_\star]}{\lambda}}.\label{eq:cauchy_seq}
 704     \end{align}
 705     We can thus construct a Cauchy sequence $\{(\theta_{t_n}, \phi_{t_n}, q_{t_n}^{1:M})\}_{n\in \mathbb{N}}$ in $\mathcal{M}^{1:M,1}$ from~\eqref{eq:cauchy_seq} for some increasing sequence $t_n \to +\infty$. By the previous Lemma~\ref{lem:cauchy}, we have the limit point $(\theta_\infty, \phi_\infty, q_\infty^{1:M}) \in \mathcal{M}_\star$. Setting $(t,t')=(0,t_n)$ in~\eqref{eq:cauchy_seq} and letting $n \to +\infty$, we have:
 706     \begin{align*}
 707         \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_0, \phi_0, q_0^{1:M}), (\theta_\infty, \phi_\infty, q_\infty^{1:M})) \leq \sqrt{\frac{2[\tilde{F}(\theta_0, \phi_0, q_0^{1:M}) - \tilde{F}_\star]}{\lambda}},
 708     \end{align*}
 709     whence the conclusion follows by noting the distance function is continuous and the infimum is attained.
 710 \end{proof}

 714 \begin{theorem}\label{thm:exp_convg_appendix}
 715     Suppose Assumptions~\Cref{assump:model_reg}-\Cref{assump:strong_log_concave} are satisfied. Then, $\tilde{\ell}$ has a unique maximizer $(\theta_\star, \phi_\star)$ and the flow converges exponentially fast to it: for some $\lambda>0$ independent of $M$, 
 716 \begin{equation}\label{eq:app_thm_exp}
 717         \mathsf{d}_{\mathcal{M}^{1:M}}((\theta_t,\phi_t,q_t^{1:M}),(\theta_\star,\phi_\star,q_{\star}^{1:M}))
 718         \leq
 719         \sqrt{\frac{2[\tilde{F}(\theta_t, \phi_t, q_t^{1:M}) - \tilde{F}_\star]}{\lambda}}
 720         \leq
 721         \sqrt{\frac{2[\tilde{F}(\theta_0, \phi_0, q_0^{1:M})-\tilde{F}_\star]}{\lambda}}e^{-\lambda t}, \quad \forall t>0;
 722 \end{equation}
 723 where $\tilde{F}_\star:=\inf_{(\theta,\phi,q^{1:M})\in \mathcal{M}^{1:M}}\tilde{F}(\theta,\phi,q^{1:M})$ and $\vert\vert\cdot\vert\vert$ denotes the Euclidean norm.
 724 \end{theorem}
 725 
 726 \begin{proof}
 727 
 728 Under Assumption~\Cref{assump:strong_log_concave}, each map $(\theta,\phi,z)\mapsto \log \tilde{p}_{\theta,\phi}(x^m,z)$ is $\lambda$-strongly concave. Hence the product density
 729 $$\bar{p}_{\theta,\phi}(x^{1:M},z^{1:M}):=\prod_{m=1}^M\tilde{p}_{\theta,\phi}(x^m,z^m)$$
 730 has joint log-density
 731 $$\bar{\ell}(\theta,\phi,z^{1:M}):=\log \bar{p}_{\theta,\phi}(x^{1:M},z^{1:M})=\sum_{m=1}^M\log\tilde{p}_{\theta,\phi}(x^m,z^m),$$
 732 which is strictly concave in $((\theta,\phi),z^{1:M})$. Applying~\citet[Theorem 4]{kuntz23a} with parameter $u=(\theta,\phi)$ and latent variable $z^{1:M}$ yields that the marginal log-likelihood
 733 $$\bar{\ell}(\theta,\phi):=\log \int \bar{p}_{\theta,\phi}(x^{1:M},z^{1:M}) \dd z^{1:M}=\sum_{m=1}^M\log\tilde{p}_{\theta,\phi}(x^m)$$
 734 has a unique maximizer. Since $\tilde{\ell}(\theta,\phi)=M^{-1}\bar{\ell}(\theta,\phi)$, the same pair $(\theta_\star,\phi_\star)$ is the unique maximizer of $\tilde{\ell}$.
 735 
 736 Under the log-concavity assumption, the family of distributions $(\tilde{\rho}_{\theta,\phi}(\dd z^{1:M}))_{(\theta,\phi)\in\Theta\times\Phi}$ satisfies the extended log-Sobolev inequality (cf. Definition~\ref{def:xLSI}) using \Cref{thm:sLC_implies_xLSI}.
 737 
 738 The rightmost inequality in~\eqref{eq:app_thm_exp} then follows from a combination of de Bruijn's identity~\eqref{eq:de_bruijin}, the extended log-Sobolev inequality, and Gr\"onwall's lemma. 
 739 
 740 The leftmost inequality follows from a combination of the uniquness of minimizers and \Cref{thm:xLSI_implies_xTalagrand}. 
 741 \end{proof}
 742 \begin{remark}\label{remark:star_notation}
 743     Since $\tilde{\ell}$ has a unique maximizer $(\theta_\star,\phi_\star)$ and Proposition~\ref{prop:ftildemins} identifies the corresponding minimizer of $\tilde{F}$, we can write $\tilde{F}_\star =\tilde{F}(\theta_\star,\phi_\star,\tilde\pi_{\theta_\star,\phi_\star}^{1:M})$ and $q_\star^{1:M}=\tilde\pi_{\theta_\star,\phi_\star}^{1:M}$, where we recall $\tilde\pi_{\theta,\phi}(\cdot)=\tilde{p}_{\theta,\phi}(\cdot|x)$ is the true posterior distribution.
 744 \end{remark}

The proof will be based on the spatial and temporal discretization error bounds established in \citet{caprioerrorboundsparticle2024}, which are combined with the exponential convergence result in Theorem~\ref{thm:exp_convg_appendix}. In what follows, we will use the notation \[\tilde\theta:=(\theta, \phi)\]s
 749 to denote the concatenation of the diffusion prior and decoder parameters as in Appendix~\ref{app:grad_deriv}.

 752 First, we recall the additional assumption.
 753 \begin{trivlist}
 754 \item[\hskip\labelsep \textbf{Assumption}~\ref{assump:lipschitz_grad} (Lipschitz gradient).]\itshape
 755 The log-likelihood $\tilde\ell^m(\tilde\theta, z):= \log p_{\tilde\theta}(x^m, z)$ is differentiable and its gradient $\nabla \tilde\ell^m := (\nabla_{\tilde\theta}\tilde\ell^m, \nabla_{z_0}\tilde\ell^m)$ is $L$-Lipschitz for some $L > 0$, that is, for all $(\tilde\theta, z), (\tilde\theta', z') \in (\Theta \times \Phi) \times \sZ$:
 756 \begin{align*}
 757     \|\nabla \tilde\ell^m(\tilde\theta, z) - \nabla \tilde\ell^m(\tilde\theta', z')\|\leq L\|(\tilde\theta, z) - (\tilde\theta', z')\|.
 758 \end{align*}
 759 \end{trivlist}
 760 
 761 We now consider the continuous-time system of SDEs that correspond to the gradient flow:
 762 \begin{align}
 763     \dd \tilde{\theta}_t &= \nabla_{\tilde\theta} \int\frac{1}{M N} \sum_{m=1}^M \sum_{n=1}^N \tilde{\ell}^m(\tilde\theta_t, z) q^{m,n}(\dd z) \, \dd t\label{eq:app-cont-theta-ode} \\
 764     \dd Z_t^{m,n} &= \nabla_{z} \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n}) \, \dd t + \sqrt{2} \, \dd W_t^{m,n}, \quad (m,n)\in[M]\times [N]\label{eq:app-cont-z-sde}
 765 \end{align}
 766 where $q_t^{m,n}:=\mathrm{Law}(Z_t^{m,n})$ is the law of the particles at time $t$. Since the $Z_t^{m,n}$ are i.i.d., we note that the first equation is equivalent to $\dd \tilde\theta = \nabla_{\tilde\theta} \int {M}^{-1}\sum_{m=1}^M \tilde{\ell}^m(\tilde\theta_t, z) q_t^m(\dd z) \, \dd t$, where $q_t^m$ is the law of $Z_t^{m,n}$ for any $n$.
 767 \begin{lemma}\label{lem:cont-time-convg}
 768     Under Assumptions~\Cref{assump:model_reg}, \Cref{assump:strong_log_concave}, and \Cref{assump:lipschitz_grad}, the system of SDEs in~\eqref{eq:app-cont-theta-ode}-\eqref{eq:app-cont-z-sde} admits a unique strong solution $(\tilde\theta_t, Z_t^{1:M,1:N})_{t \geq 0}$ and the solution satisfies
 769     \begin{equation*}
 770         \mathsf{d}_{\mathcal{M}^{1:M}}((\tilde\theta_t, Q_t^{1:M,N}), (\tilde\theta_\star, Q_\star^{1:M,N})) \leq C e^{-\lambda t}, \quad \forall t>0,
 771     \end{equation*}
 772     where $C>0$ is a constant independent of $N$ and $t$, and we defined the empirical distributions to $q_t^m$ and $q_\star^m$ as $Q_t^{m,N}:=N^{-1}\sum_{n=1}^N \delta_{Z_t^{m,n}}$ and $Q_\star^{m,N}:=N^{-1}\sum_{n=1}^N \delta_{Z_\star^{m,n}}$ respectively, where $Z_\star^{m,n}$ are i.i.d. samples from $\tilde\pi_{\tilde\theta_\star}$ (cf. Remark~\ref{remark:star_notation}).
 773 \end{lemma}
 774 \begin{proof}
 775     The existence and uniqueness of a strong solution follows from Proposition 8 in  \citet{caprioerrorboundsparticle2024} by replacing the system of SDEs therein with ours. To prove the inequality, we note that by a coupling argument for each pair $(Z_t^{m,n}, Z_\star^{m,n})$, we have:
 776     \begin{align*}
 777         \mathsf{d}_{\mathcal{M}^{1:M}}((\tilde\theta_t, Q_t^{1:M,N}), (\tilde\theta_\star, Q_\star^{1:M,N}))^2 &\leq \|\tilde\theta_t-\tilde\theta_\star\|^2 + \frac{1}{MN} \sum_{m=1}^{M}\sum_{n=1}^{N} \mathbb{E}[\|Z_t^{m,n}-Z_\star^{m,n}\|^2] \\
 778         &=\|\tilde\theta_t-\tilde\theta_\star\|^2 + \frac1M \sum_{m=1}^{M}\mathbb{E}[\|Z_t^{m,1}-Z_\star^{m,1}\|^2] \\
 779         &= \mathsf{d}_{\mathcal{M}^{1:M}}((\tilde\theta_t, q_t^{1:M}), (\tilde\theta_\star, q_\star^{1:M}))^2 \leq \frac{2e^{-2\lambda t}}{\lambda} [\tilde{F}(\tilde\theta_0, q_0) - \tilde{F}_\star],
 780     \end{align*}
 781     where we used~\Cref{thm:exp_convg_appendix} in the last inequality.
 782 \end{proof}

 786 \begin{lemma}[Spatial Discretization Error]\label{lem:spatial_discretization}
 787     Suppose Assumptions~\Cref{assump:strong_log_concave} and~\Cref{assump:lipschitz_grad} hold. Then the following system of SDEs:
 788     \begin{align}
 789         \dd \tilde\theta_t^N &= \frac{1}{MN} \sum_{m=1}^{M} \sum_{n=1}^N \nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) \, \dd t \label{eq_app:spatial1} \\
 790         \dd \bar{Z}_t^{m,n} &= \nabla_{z} \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) \, \dd t + \sqrt{2} \, \dd W_t^{m,n}, \quad (m,n)\in[M]\times [N], \label{eq_app:spatial3}
 791     \end{align}
 792     has a strong solution $(\tilde\theta_t^N, \bar{Z}_t^{1:M, 1:N})_{t \geq 0}$. Furthermore, there exists a constant $C(N)>0$ of order $\mathcal{O}(N^{-1/2})$ independent of $t$ and $M$, such that:
 793     \begin{equation*}
 794         \mathsf{d}_{\mathcal{M}^{1:M}}((\tilde\theta_t^N, \bar{Q}_t^{1:M, N}), (\tilde\theta_t, Q_t^{1:M,N})) \leq C(N), \quad \forall t>0,
 795     \end{equation*}
 796     where $\bar{Q}_t^{m, N}:=N^{-1}\sum_{n=1}^N \delta_{\bar{Z}_t^{m,n}}$ for each $m\in [M]$ and $Q_t^{m,N}$ is defined as in Lemma~\ref{lem:cont-time-convg}.
 797 \end{lemma}
 798 \begin{proof}
 799     The proof is a modification of~\citet[Lemma 13]{caprioerrorboundsparticle2024} by replacing the SDEs therein with \eqref{eq_app:spatial1}-\eqref{eq_app:spatial3}. More specifically, we define the quantity $\xi^N_t$ as:
 800     \begin{equation}
 801         \xi_t^N := \|\tilde\theta_t^N - \tilde\theta_t\|^2 + \frac1M \sum_{m=1}^M \frac1N \sum_{n=1}^N \|Z_t^{m,n} - \bar{Z}_t^{m,n}\|^2,
 802     \end{equation}
 803     and we will show that $\mathbb{E}[\xi_t^N] \leq C(N)$ for some constant $C(N)$ which upper bounds the spatial discretization error. Using Itô's formula:
 804     \begin{align}
 805         \dd \|\tilde\theta_t^N - \tilde\theta_t\|^2 &= 2 \left\langle \tilde\theta_t^N - \tilde\theta_t, \frac{1}{MN} \sum_{m=1}^{M} \sum_{n=1}^N \nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) - \nabla_{\tilde\theta} \int \frac1M \sum_{m=1}^M \tilde{\ell}^m(\tilde\theta_t, z) q_t^m(\dd z) \right\rangle \dd t, \label{eq_app:spatial_error_theta}\\
 806         \dd \|\bar{Z}_t^{m,n} - {Z}_t^{m,n}\|^2 &= 2 \left\langle  
 807             \bar{Z}_t^{m,n} - {Z}_t^{m,n}, \nabla_z \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) - \nabla_z \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n})
 808         \right\rangle \dd t.\label{eq_app:spatial_error_z}
 809     \end{align}
 810     Adding and subtracting $(MN)^{-1}\sum_m\sum_n\nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n})$ from~\eqref{eq_app:spatial_error_theta} and summing with~\eqref{eq_app:spatial_error_z} scaled by $(MN)^{-1}$ yields:
 811     \begin{align*}
 812         \dd \xi_t^N &= \frac{2}{MN} \sum_{m=1}^M \sum_{n=1}^N 2\bigl[\left\langle \tilde\theta_t^N - \tilde\theta_t, \nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) - \nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n}) 
 813         \right\rangle \dd t \\
 814         &+\left\langle \bar{Z}_t^{m,n} - {Z}_t^{m,n}, \nabla_z \tilde{\ell}^m(\tilde\theta_t^N, \bar{Z}_t^{m,n}) - \nabla_z \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n})\right\rangle
 815         \bigr] \dd t + 2 G_t^N \dd t,
 816     \end{align*}
 817     where we defined the term $G_t^N$ as:
 818     \begin{align*}
 819         G_t^N &:= \frac1M \sum_{m=1}^M G_t^{m,N}, \quad \text{where}\  G_t^{m,N} := \left\langle \tilde\theta_t^N - \tilde\theta_t, \frac1N \sum_{n=1}^{N} \nabla_{\tilde\theta} \tilde{\ell}^m(\tilde\theta_t, Z_t^{m,n}) - \nabla_{\tilde\theta} \int \tilde{\ell}^m(\tilde\theta_t, z) q_t^m(\dd z) \right\rangle.
 820     \end{align*}
 821     Using the strong log-concavity~\Cref{assump:strong_log_concave}, 
 822     and taking the expectation yields
 823     \begin{align*}
 824         \dd \mathbb{E}[\xi_t^N] &\leq -2\lambda \mathbb{E}[\xi_t^N] \dd t + 2 \mathbb{E}[G_t^N] \dd t.
 825     \end{align*}
 826     Following the same argument as in~\citet[Lemma 13]{caprioerrorboundsparticle2024}, together with the analogous second-moment bound from~\citet[Proposition 26]{caprioerrorboundsparticle2024} and the $L$-Lipschitz gradient assumption~\Cref{assump:lipschitz_grad}, we can upper bound $|\mathbb{E}[G_t^{m,N}]|$ for each $m\in [M]$:
 827     \begin{equation}
 828         \left|\mathbb{E}[G_t^{m,N}]\right|\leq L \sqrt{\frac{2\mathbb{E}\left[\|\tilde\theta_t^N - \tilde\theta_t\|^2\right]}{N} \left(\|\tilde\theta_0\|^2 + \mathbb{E}[\|Z_0\|^2] + \frac{d_z}{\lambda}\right)}\leq L \sqrt{\frac{2\mathbb{E}[\xi_t^N]}{N} \left(\|\tilde\theta_0\|^2 + \mathbb{E}[\|Z_0\|^2] + \frac{d_z}{\lambda}\right)}.
 829     \end{equation}
 830     Now we have the following differential inequality for $\mathbb{E}[\xi_t^N]$:
 831     \begin{equation}
 832         \frac{\dd}{\dd t} \mathbb{E}[\xi_t^N]^{1/2} \leq -\lambda \mathbb{E}[\xi_t^N]^{1/2} + L \sqrt{\frac{2}{N} \left(\|\tilde\theta_0\|^2 + \mathbb{E}[\|Z_0\|^2] + \frac{d_z}{\lambda}\right)}.
 833     \end{equation}
 834     Applying Gr\"onwall's lemma yields the result:
 835     \begin{equation}
 836         \mathbb{E}[\xi_t^N]^{1/2} \leq e^{-\lambda t} \mathbb{E}[\xi_0^N]^{1/2} + \frac{(1-e^{-\lambda t})L}{\lambda} \sqrt{\frac{2}{N} \left(\|\tilde\theta_0\|^2 + \mathbb{E}[\|Z_0\|^2] + \frac{d_z}{\lambda}\right)},
 837     \end{equation}
 838     where the first term is zero due to construction.
 839 \end{proof}

 841 \begin{lemma}[Temporal Discretization Error]\label{lem:temporal_discretization}
 842     Suppose Assumptions~\Cref{assump:strong_log_concave} and~\Cref{assump:lipschitz_grad} hold. Then for the following Euler-Maruyama scheme for the SDEs in \eqref{eq_app:spatial1}-\eqref{eq_app:spatial3}:
 843     \begin{align}
 844         \tilde\theta_{k+1}^{N,h} &= \tilde\theta_k^{N,h} + \frac{h}{MN} \sum_{m=1}^{M}\sum_{n=1}^N \nabla_{\tilde\theta} \tilde{\ell}(\tilde\theta_k^{N,h}, \bar{Z}_k^{m,n,N,h}) \label{eq_app:temp1} \\
 845         \bar{Z}_{k+1}^{m,n,N,h} &= \bar{Z}_k^{m,n,N,h} + h \nabla_{z} \tilde{\ell}(\tilde\theta_k^{N,h}, \bar{Z}_k^{m,n,N,h}) + \sqrt{2h} W_k^n, \quad (m,n) \in [M] \times [N], \label{eq_app:temp3}
 846     \end{align}
 847     we have for $h \leq {1}/({\lambda+L})$ with $\lambda$ being the strong concavity constant in Assumption~\Cref{assump:strong_log_concave} and $L$ being the Lipschitz constant in \Cref{assump:lipschitz_grad}, there exists a constant $C(h)>0$ independent of $k$ and $N$ of order $\mathcal{O}(h^{1/2})$,
 848     such that:
 849     \begin{equation*}
 850         \mathsf{d}_{\mathcal{M}^{1:M}}((\tilde\theta_k^{N,h}, \bar{Q}_k^{1:M,N,h}), (\tilde\theta_{kh}^N, \bar{Q}_{kh}^{1:M,N})) \leq \sqrt{h}C(h), \quad \forall k \in \mathbb{N},
 851     \end{equation*}
 852 \end{lemma}
 853 \begin{proof}
 854     The proof is a direct adaptation of~\citet[Lemma 14]{caprioerrorboundsparticle2024} by replacing the discretization therein with~\cref{eq_app:temp1,eq_app:temp3} and using the same argument for the $M\times N$ equations for the particles and replacing the loss with $M^{-1}\sum_{m=1}^M \tilde{\ell}^m$ for the parameters.
 855 \end{proof}

 876 \subsection{Evolution of the Particle Cloud}
 877 We also visualize how the particle cloud $z_{0}^{1:M, 1:N}$ evolves through the gradient flow. To achieve this, we save the particle cloud at training epochs $\{0,5,10,15,20,25,30,50,75,90\}$ and pass them through the decoder $g_\phi$ (\Cref{fig:evo_cifar,fig:evo_celeba}). We note that there is an intriguing perceptual similarity between the particle evolution and the posterior mean $\mathbb{E}[ x_0 | x_t]$ of a diffusion model. For comparison, we show the evolution of the posterior mean predicted in \Cref{fig:evo_ddim_cifar,fig:evo_ddim_celeba}. The posterior mean is computed by a pretrained score network taken from \citet{song2022denoisingdiffusionimplicitmodels} in the pixel space via Tweedie's formula \citep{efron2011tweedie}: $\mathbb{E}[ x_0| x_t] \approx  (\sqrt{\bar{\alpha}_t})^{-1} [  x_t + \sigma_t^2 \nabla_{ x_t} \vs_\theta( x_t, t)]$, where $\sigma_t^2$ is the variance of the reverse process and $\vs_\theta( x_t, t)$ is the score estimating network.
 878 We remark that there have been several recent works attempting to delineate the connection between diffusion and gradient flow \citep{yiMonoFlowRethinkingDivergence2023,huangGANsGradientFlows2023,franceschiUnifyingGANsScorebased2023}. Our method can be thought as an attempt of learning the gradient flow via a diffusion model.

 931 \subsection{Ablation on Two Stage Training}
 932 \label{app:2stage_ablate}
 933 \citet{leng2025repaeunlockingvaeendtoend} hypothesized that backpropogating the diffusion loss to the \gls*{VAE} directly make the latent space simpler. They suggested this could inadvertently "hack" the denoising objective, leading the diffusion model to simply predict noise from the \gls*{VAE}'s Gaussian approximate posterior.
 934 
 935 To investigate this, we conduct an ablation study comparing end-to-end training and two-stage training. We train both models with $1$ particle using the same training hyperparameters; for two-stage training we detach the gradients on the particles before passing them to the diffusion model, hence preventing backpropogating the diffusion loss. We show in \Cref{tab:app_ablate_2stage} that end-to-end training in \gls*{IPLD} does not suffer from the same issue observed in \citet{leng2025repaeunlockingvaeendtoend}. This aligns with the results in \citet{vahdatscorebasedgenerativemodeling2021}, who also found benefits to end-to-end training. It is important to note, however, that both our experiments and those of \citet{vahdatscorebasedgenerativemodeling2021} were conducted on relatively lower-resolution datasets. Therefore, the conclusions drawn here may not directly extrapolate to the larger-scale experimental settings used in \citet{leng2025repaeunlockingvaeendtoend}.
