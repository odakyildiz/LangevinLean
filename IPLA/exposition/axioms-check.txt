@IPLA.theorem1 : ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (P : MeasureTheory.Measure Ω)
  [inst_1 : MeasureTheory.IsProbabilityMeasure P] (N dθ dx : ℕ),
  0 < N →
    ∀ {Data : Type u_2} (U : Data → IPLA.Joint dθ dx → ℝ) (y : Data) (m γ : ℝ) (L : NNReal)
      (hdiff : Differentiable ℝ (U y)),
      StrongConvexOn Set.univ m (U y) →
        ∀ (hLip : LipschitzWith L (gradient (U y))),
          0 < m →
            0 < γ →
              γ ≤ 2 / (m + ↑L) →
                ∀ (θStar : EuclideanSpace ℝ (Fin dθ)),
                  IPLA.IsMarginalMaximizer (U y) θStar →
                    ∀ (initial : ↥(MeasureTheory.Lp (IPLA.State N dθ dx) 2 P))
                      (noise : ℕ → ↥(MeasureTheory.Lp (IPLA.State N dθ dx) 2 P)),
                      IPLA.GaussianInnovations P initial noise →
                        ∀ (k : ℕ),
                          √(∫ (z : IPLA.State N dθ dx),
                                ‖IPLA.parameterCLM z - θStar‖ ^ 2 ∂IPLA.laws P (U y) γ L hdiff hLip initial noise k) ≤
                            √(2 * ↑dθ / (↑N * m)) +
                                (1 - m * γ) ^ k *
                                  ULA.wasserstein2Real (MeasureTheory.Measure.map (↑↑initial) P)
                                    (IPLA.gibbsTarget (U y)) +
                              33 / 20 * (↑L / m) * √(↑dx + ↑dθ / ↑N) * √γ
